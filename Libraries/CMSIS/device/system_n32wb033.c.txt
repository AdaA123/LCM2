/**
*\*\name   SystemInit.
*\*\fun    Setup the microcontroller system. Initialize the Embedded Flash Interface,
*\*\        the PLL and update the SystemCoreClock variable.
*\*\param  none
*\*\return none
**/
/* This function should be used only after reset. */
void SystemInit(void)
{
    uint32_t tmp;
	
		*(volatile uint8_t *)(0x2000007CU) =  92;
		*(volatile uint8_t *)(0x2000007DU) =  12;

    __set_PRIMASK(0U); /* enable irq */
    SCB->VTOR = 0x01000000U;

    RCC->APB1PCLKEN |= RCC_APB1PCLKEN_PWREN; /* PWR enable */

    *(volatile uint32_t*)(uintptr_t)0x40007014U = 0x0000080FU; /* VDDD=1.0V */
    *(volatile uint32_t*)(uintptr_t)0x4002245CU = 0x09U;       /* fsynldo = 1.0V */
    *(volatile uint32_t*)(uintptr_t)0x40007020U = 0x00020018U;
	*(volatile uint32_t*)(uintptr_t)0x40011800U &= ~0xF0000U;
	*(volatile uint32_t*)(uintptr_t)0x40011800U |= 0x30000U;

    SystemTrimValueRead((uint8_t*)&trim_stored, sizeof(trim_stored));

    /* check otp has been written */
    if ((trim_stored.stote_rc64m_trim_value == 0xFFFFFFFFU) || (trim_stored.stote_rc64m_trim_value == 0U))
    {
        RCC->CFG |= RCC_CFG_HSISRC; /* USE HSI as system clock */
        RCC->CFG &= ~RCC_CFG_APB1PRES;
        RCC->CFG |= RCC_CFG_APB1PRES_2; /* APB1 = HCLK/2, APB1 max is 32M */
        /* Calib from HSE */
        RCC_HsiCalib(SYSCLK_FREQ);
    }
    else
    {
        tmp = (PWR->reserved4) & (~0x1FU);
        tmp |= (trim_stored.stote_bg_vtrim_value) & 0x1FU;
        PWR->reserved4 = tmp;
        if (SYSCLK_FREQ == 64000000U)
        {
            RCC->CTRL &= ~0x8000U; /* Set HSI as 64M */
            /* Configures LSI trim */
            tmp = RCC->CTRL & ~((uint32_t)0x7FU << 8U); /* TRIM 8-14 bit */
            RCC->CTRL = tmp | (trim_stored.stote_rc64m_trim_value << 8U); /* clear and set TRIM value */

            RCC->CFG |= RCC_CFG_HSISRC; /* USE HSI as system clock */

            RCC->CFG &= ~RCC_CFG_APB1PRES;
            RCC->CFG |= RCC_CFG_APB1PRES_2; /* APB1 = HCLK/2, APB1 max is 32M */
        }

        /* Configures LSI trim */
        RCC->LSCTRL &= ~RCC_LSCTRL_LSITRIM;
        RCC->LSCTRL |= trim_stored.stote_rc32768_trim_value << 8U;
    }
}

/**
*\*\name   System_Init.
*\*\fun    Update SystemCoreClock variable according to Clock Register Values.
*\*\        The SystemCoreClock variable contains the core clock (HCLK), it can
*\*\        be used by the user application to setup the SysTick timer or
*\*\        configure other parameters.
*\*\param  none
*\*\return none
**/
/* Each time the core clock (HCLK) changes, this function must be called to update
 * SystemCoreClock variable value. Otherwise, any configuration based on this variable
 * will be incorrect.
 * The system frequency computed by this function is not the real frequency in the chip.
 * It is calculated based on the predefined constant and the selected clock source. */
void System_Init(void)
{
    SystemCoreClock = HSI_VALUE;
    #include "flash.h"
    SCB->VTOR = APP_ADDRESS;
}

/**
*\*\name   system_delay_cycles.
*\*\fun    Delay cycles.
*\*\param  i :
*\*\          cycle count.
*\*\return none
**/
void system_delay_cycles(uint32_t i)
{
    volatile uint32_t count = i;
    while (count-- != 0U)
    {
        __asm("nop");
    }
}

/**
*\*\name   system_delay_n_10us.
*\*\fun    Delay n * 10 us.
*\*\param  value :
*\*\          delay count in 10us units.
*\*\return none
**/
void system_delay_n_10us(uint32_t value)
{
    system_delay_cycles(107U * value);
}

/**
*\*\name   RCC_HsiCalib.
*\*\fun    Enable the HSI and calibrate it.
*\*\param  systemfreq :
*\*\          target system frequency.
*\*\return none
**/
static void RCC_HsiCalib(uint32_t systemfreq)
{
    uint32_t g_hsi_accuracy = 0U;
    uint32_t g_timeoutcnt = 1000U;
    uint32_t g_cal_hsi_cnt_value = 1024U;

    uint32_t delta = 0U;
    uint32_t min = 0U;
    uint32_t max = 127U;                                /* 32M TRIM 8-14 bit */
    uint32_t mid = 64U;
    uint32_t count_value = 0U;
    uint32_t hsi_timeoutcnt = 0U;
    uint32_t tmp_trim;
    uint32_t min_delta_trim = 0U;
    uint32_t min_delta = 1024U;
    uint32_t tmp = 0U;

    if (systemfreq <= 64000000U)
    {
        RCC->CTRL &= ~0x8000U;
    }
    else
    {
        RCC->CTRL |= 0x8000U;
    }

    g_cal_hsi_cnt_value = systemfreq / 1000U / 1000U * 32U;

    do
    {
        tmp = RCC->CTRL & ~((uint32_t)0x7FU << 8U);               /* 32M TRIM 8-14 bit */
        RCC->CTRL = tmp | (mid << 8U);                  /* clear and set TRIM value and start to cnt */
        system_delay_cycles(5U);                        /* delay scape */
        while (1)
        {
            system_delay_cycles(1U);
            if (((RCC->OSCFCSR & 0x02U) != 0U) || (hsi_timeoutcnt++ > g_timeoutcnt))
            {
                break;
            }
        }
        count_value = RCC->OSCFCHSICNT;                 /* ready cnt value */
        if (count_value > g_cal_hsi_cnt_value)
        {
            delta = count_value - g_cal_hsi_cnt_value;
        }
        else
        {
            delta = g_cal_hsi_cnt_value - count_value;
        }

        if (delta < min_delta)
        {
            min_delta_trim = mid;
            min_delta = delta;
        }

        if (count_value >= g_cal_hsi_cnt_value)
        {
            max = mid;
        }
        else
        {
            min = mid;
        }

        tmp_trim = (min + max) / 2U;
        if (tmp_trim == mid)                            /* 0 and 127 if not used */
        {
            break;
        }
        mid = tmp_trim;
    } while (delta > g_hsi_accuracy);

    tmp = RCC->CTRL & ~((uint32_t)0x7FU << 8U);                  /* 32M TRIM 8-14 bit */
    RCC->CTRL = tmp | (min_delta_trim << 8U);           /* clear and set TRIM value and start to cnt */

    RCC->CFG &= ~1U;
    while ((RCC->CFG & (1U << 2U)) != 0U)
    {
        /* wait for clock switch */
    }
}
