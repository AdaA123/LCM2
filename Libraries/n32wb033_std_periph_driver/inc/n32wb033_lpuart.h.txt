#ifndef __N32WB033_LPUART_H__
#define __N32WB033_LPUART_H__

#ifdef __cplusplus
extern "C" {
#endif

#include "n32wb033.h"

/** LPUART Init Structure definition **/

typedef struct
{
    uint32_t BaudRate;            /* This member configures the LPUART communication baud rate.
                                           The baud rate is computed using the following formula:
                                            - IntegerDivider = ((CLK) / (LPUART_InitStruct->BaudRate))
                                            - FractionalDivider */

    uint32_t Parity;              /* Specifies the parity mode. */

    uint32_t Mode;                /* Specifies wether the Receive or Transmit mode is enabled or disabled. */

    uint32_t RtsThreshold;        /* Specifies RTS Threshold. */

    uint32_t HardwareFlowControl; /* Specifies wether the hardware flow control mode is enabled or disabled. */
} LPUART_InitType;

/** LPUART Structure Definition End **/

/** LPUART Macro Definition Start **/

#define LPUART_REG_BIT_MASK        (uint32_t)0x00000000U

/* LPUART CTRL Clear Mask */
#define LPUART_CTRL_CLR_MASK           ((uint32_t)(~(LPUART_CTRL_TXEN \
                                              | LPUART_CTRL_PSEL \
                                              | LPUART_CTRL_PCDIS \
                                              | LPUART_CTRL_RTS_THSEL_0 \
                                              | LPUART_CTRL_RTS_THSEL_1 \
                                              | LPUART_CTRL_CTSEN \
                                              | LPUART_CTRL_RTSEN)))

/** Configures the parity of LPUART **/
#define LPUART_PE_NO               ((uint32_t)LPUART_CTRL_PCDIS)               /* Parity disable */
#define LPUART_PE_EVEN             ((uint32_t)LPUART_REG_BIT_MASK)             /* Even parity */
#define LPUART_PE_ODD              ((uint32_t)LPUART_CTRL_PSEL)                /* Odd parity */

/** LPUART is configured as RX or TX **/
#define LPUART_MODE_RX             ((uint32_t)LPUART_REG_BIT_MASK)             /* Receiver enable */
#define LPUART_MODE_TX             ((uint32_t)LPUART_CTRL_TXEN)                /* Transmitter enable */

/** LPUART RTS Threshold selection **/
#define LPUART_RTSTH_FIFOHF        ((uint32_t)LPUART_REG_BIT_MASK)             /* FIFO Half Full */
#define LPUART_RTSTH_FIFO3QF       ((uint32_t)LPUART_CTRL_RTS_THSEL_0)         /* FIFO 3/4 Full */
#define LPUART_RTSTH_FIFOFU        ((uint32_t)LPUART_CTRL_RTS_THSEL_1)         /* FIFO Full */

/** LPUART Hardware Flow Control **/
#define LPUART_HFCTRL_NONE         ((uint32_t)LPUART_REG_BIT_MASK)
#define LPUART_HFCTRL_RTS          ((uint32_t)LPUART_CTRL_RTSEN)
#define LPUART_HFCTRL_CTS          ((uint32_t)LPUART_CTRL_CTSEN)
#define LPUART_HFCTRL_RTS_CTS      ((uint32_t)(LPUART_CTRL_RTSEN | LPUART_CTRL_CTSEN))

/** LPUART Interrupt definition **/
#define LPUART_INT_MASK                ((uint32_t)0x0000007FU) /* LPUART Interrupt internal Mask */
#define LPUART_INT_PEF                 ((uint32_t)(((uint32_t)0x00U << 8U) | LPUART_INTEN_PEIE))       /* PEIE=bit0 */
#define LPUART_INT_TXC                 ((uint32_t)(((uint32_t)0x01U << 8U) | LPUART_INTEN_TXCIE))      /* TXCIE=bit1 */
#define LPUART_INT_FIFO_OV             ((uint32_t)(((uint32_t)0x02U << 8U) | LPUART_INTEN_FIFO_OVIE))  /* FIFO_OVIE=bit2 */
#define LPUART_INT_FIFO_FU             ((uint32_t)(((uint32_t)0x03U << 8U) | LPUART_INTEN_FIFO_FUIE))  /* FIFO_FUIE=bit3 */
#define LPUART_INT_FIFO_HF             ((uint32_t)(((uint32_t)0x04U << 8U) | LPUART_INTEN_FIFO_HFIE))  /* FIFO_HFIE=bit4 */
#define LPUART_INT_FIFO_NE             ((uint32_t)(((uint32_t)0x05U << 8U) | LPUART_INTEN_FIFO_NEIE))  /* FIFO_NEIE=bit5 */
#define LPUART_INT_WUF                 ((uint32_t)(((uint32_t)0x06U << 8U) | LPUART_INTEN_WUFIE))      /* WUFIE=bit6 */

/** LPUART DMA Requests **/
#define LPUART_DMAREQ_TX             ((uint32_t)LPUART_CTRL_DMA_TXEN)
#define LPUART_DMAREQ_RX             ((uint32_t)LPUART_CTRL_DMA_RXEN)

/** LPUART WakeUp methods **/
#define LPUART_WUSEL_CLR_MASK          ((uint32_t)(~LPUART_CTRL_WUSEL))
#define LPUART_WUSTP_STARTBIT          ((uint32_t)LPUART_REG_BIT_MASK)
#define LPUART_WUSTP_RXNE              ((uint32_t)LPUART_CTRL_WUSEL_0)
#define LPUART_WUSTP_BYTE              ((uint32_t)LPUART_CTRL_WUSEL_1)
#define LPUART_WUSTP_FRAME             ((uint32_t)(LPUART_CTRL_WUSEL_0 | LPUART_CTRL_WUSEL_1))

/** LPUART WakeUp mode in Sleep enable **/
#define CTRL_WUSTP_SET                 ((uint32_t)LPUART_CTRL_WUSTP)
#define CTRL_WUSTP_RESET               ((uint32_t)(~LPUART_CTRL_WUSTP))

/** LPUART Sampling methods **/
#define LPUART_SMPCNT_CLR_MASK         ((uint32_t)(~LPUART_CTRL_SMPCNT))
#define LPUART_SMPCNT_3B               ((uint32_t)LPUART_REG_BIT_MASK) /* 3 Sample bit */
#define LPUART_SMPCNT_1B               ((uint32_t)LPUART_CTRL_SMPCNT)  /* 1 Sample bit */

/** LPUART LoopBack Self-Test **/
#define CTRL_LOOPBACK_SET              ((uint32_t)LPUART_CTRL_LOOKBACK)
#define CTRL_LOOPBACK_RESET            ((uint32_t)(~LPUART_CTRL_LOOKBACK))

/** LPUART Flush Receiver FIFO **/
#define CTRL_FLUSH_SET                 ((uint32_t)LPUART_CTRL_FLUSH)
#define CTRL_FLUSH_RESET               ((uint32_t)(~LPUART_CTRL_FLUSH))

/** LPUART Flags **/
#define LPUART_FLAG_PEF                ((uint32_t)LPUART_STS_PEF)
#define LPUART_FLAG_TXC                ((uint32_t)LPUART_STS_TXC)
#define LPUART_FLAG_FIFO_OV            ((uint32_t)LPUART_STS_FIFO_OV)
#define LPUART_FLAG_FIFO_FU            ((uint32_t)LPUART_STS_FIFO_FU)
#define LPUART_FLAG_FIFO_HF            ((uint32_t)LPUART_STS_FIFO_HF)
#define LPUART_FLAG_FIFO_NE            ((uint32_t)LPUART_STS_FIFO_NE)
#define LPUART_FLAG_CTS                ((uint32_t)LPUART_STS_CTS)
#define LPUART_FLAG_WUF                ((uint32_t)LPUART_STS_WUF)
#define LPUART_FLAG_NF                 ((uint32_t)LPUART_STS_NF)

/* Reset and init */
void LPUART_DeInit(const LPUART_Module* LPUARTx);
void LPUART_Init(LPUART_Module* LPUARTx, const LPUART_InitType* LPUART_InitParam);
void LPUART_StructInit(LPUART_InitType* LPUART_InitParam);

/* FIFO */
void LPUART_FlushRxFifo(LPUART_Module* LPUARTx);

/* Data transfer */
void LPUART_SendData(LPUART_Module* LPUARTx, uint8_t Data);
uint8_t LPUART_ReceiveData(const LPUART_Module* LPUARTx);

/* WakeUp configuration */
void LPUART_ConfigWakeUpMethod(LPUART_Module* LPUARTx, uint32_t LPUART_WakeUpMethod);
void LPUART_EnableWakeUpSleep(LPUART_Module* LPUARTx, FunctionalState Cmd);
void LPUART_ConfigWakeUpData(LPUART_Module* LPUARTx, uint32_t LPUART_WakeUpData);

/* Miscellaneous configuration */
void LPUART_ConfigSamplingMethod(LPUART_Module* LPUARTx, uint32_t LPUART_SamplingMethod);
void LPUART_EnableLoopBack(LPUART_Module* LPUARTx, FunctionalState Cmd);

/* DMA */
void LPUART_EnableDMA(LPUART_Module* LPUARTx, uint32_t LPUART_DMAReq, FunctionalState Cmd);

/* Interrupt and flag management */
void LPUART_ConfigInt(LPUART_Module* LPUARTx, uint32_t LPUART_INT, FunctionalState Cmd);
FlagStatus LPUART_GetFlagStatus(const LPUART_Module* LPUARTx, uint32_t LPUART_FLAG);
void LPUART_ClrFlag(LPUART_Module* LPUARTx, uint32_t LPUART_FLAG);
INTStatus LPUART_GetIntStatus(const LPUART_Module* LPUARTx, uint32_t LPUART_INT);
void LPUART_ClrIntPendingBit(LPUART_Module* LPUARTx, uint32_t LPUART_INT);

#ifdef __cplusplus
}
#endif

#endif /* __N32WB033_LPUART_H__ */
