#ifndef __N32WB033_IRC_H__

#include <stdbool.h>
#include <stdint.h>
#include "n32wb033.h"

/** IR logic bit format **/
typedef enum {
    HW_IRGEN_LOGIC_FORMAT_MARK = 0,
    HW_IRGEN_LOGIC_FORMAT_SPACE = 1
} HW_IRGEN_LOGIC_FORMAT;

/** IR FIFO **/
typedef enum {
    HW_IRGEN_FIFO_CODE = 0,
    HW_IRGEN_FIFO_REPEAT = 1
} HW_IRGEN_FIFO;

/** Paint symbol **/
typedef enum {
    HW_IRGEN_PAINT_SPACE = 0,
    HW_IRGEN_PAINT_MARK = 1
} HW_IRGEN_PAINT;

/** IR generator output mode **/
typedef enum {
    HW_IRGEN_OUTPUT_NORMAL = 0,
    HW_IRGEN_OUTPUT_INVERTED = 1
} HW_IRGEN_OUTPUT;

/** IRC configuration **/

/**
*\*\name    hw_irgen_set_carrier_freq.
*\*\fun     Set carrier frequency.
*\*\param   hi_time :
*\*\          - carrier signal high duration in IR clock cycles (must be >0)
*\*\param   lo_time :
*\*\          - carrier signal low duration in IR clock cycles (must be >0)
*\*\return  none
**/
static __INLINE void hw_irgen_set_carrier_freq(uint16_t hi_time, uint16_t lo_time)
{
    IRC->FREQ_CARRIER_ON = hi_time;
    IRC->FREQ_CARRIER_OFF = lo_time;
}

/**
*\*\name    hw_irgen_set_logic0_param.
*\*\fun     Set "logic 0" parameters.
*\*\param   format :
*\*\          - "logic 0" format
*\*\param   mark_time :
*\*\          - mark duration in carrier clock cycles (must be >0)
*\*\param   space_time :
*\*\          - space duration in carrier clock cycles (must be >0)
*\*\return  none
**/
static __INLINE void hw_irgen_set_logic0_param(HW_IRGEN_LOGIC_FORMAT format, uint8_t mark_time,
                                                                                uint8_t space_time)
{
    IRC->LOGIC_ZERO_TIME = (uint16_t)(mark_time << IR_LOGIC_ZERO_TIME_LOGIC_ZERO_MARK_Pos | space_time);
    IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_LOGIC_ZERO_FORMAT_Msk)) | \
        ((IR_CTRL_LOGIC_ZERO_FORMAT_Msk) & ((format) << (IR_CTRL_LOGIC_ZERO_FORMAT_Pos))));
}

/**
*\*\name    hw_irgen_set_logic1_param.
*\*\fun     Set "logic 1" parameters.
*\*\param   format :
*\*\          - "logic 1" format
*\*\param   mark_time :
*\*\          - mark duration in carrier clock cycles (must be >0)
*\*\param   space_time :
*\*\          - space duration in carrier clock cycles (must be >0)
*\*\return  none
**/
static __INLINE void hw_irgen_set_logic1_param(HW_IRGEN_LOGIC_FORMAT format, uint8_t mark_time,
                                                                                uint8_t space_time)
{
    IRC->LOGIC_ONE_TIME = (uint16_t)(mark_time << IR_LOGIC_ONE_TIME_LOGIC_ONE_MARK_Pos | space_time);
    IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_LOGIC_ONE_FORMAT_Msk)) | \
        ((IR_CTRL_LOGIC_ONE_FORMAT_Msk) & ((format) << (IR_CTRL_LOGIC_ONE_FORMAT_Pos))));
}

/**
*\*\name    hw_irgen_set_repeat_time.
*\*\fun     Set command repeat time.
*\*\param   time :
*\*\          - repeat time in carrier clock cycles
*\*\return  none
**/
static __INLINE void hw_irgen_set_repeat_time(uint16_t time)
{
    IRC->REPEAT_TIME = time;
}

/** IRC FIFO management **/

/**
*\*\name    hw_irgen_insert_digital_message.
*\*\fun     Insert Digital message into FIFO.
*\*\        Digital message is represented by payload as sequence of "logic 0" and "logic 1".
*\*\param   fifo :
*\*\          - FIFO to be used
*\*\param   length :
*\*\          - number of valid bits in payload (must be >0)
*\*\param   payload :
*\*\          - message content
*\*\return  none
**/
static __INLINE void hw_irgen_insert_digital_message(HW_IRGEN_FIFO fifo, uint8_t length, uint16_t payload)
{
    /* reg[20] = digital message (1), reg[19:16] = length-1, reg[15:0] = payload */
    uint32_t value = 0x100000 | (((length - 1) & 0xF) << 16) | (payload & 0xFFFF);
    if (fifo == HW_IRGEN_FIFO_REPEAT) {
        IRC->REPEAT_FIFO = value;
    } else {
        IRC->CODE_FIFO = value;
    }
}

/**
*\*\name    hw_irgen_insert_paint_message.
*\*\fun     Insert Paint message into FIFO.
*\*\        Paint message is represented by either mark or space and its duration.
*\*\        This allows to represent any custom "painted" waveform.
*\*\param   fifo :
*\*\          - FIFO to be used
*\*\param   symbol :
*\*\          - symbol type
*\*\param   duration :
*\*\          - symbol duration in carrier clock cycles
*\*\return  none
**/
static __INLINE void hw_irgen_insert_paint_message(HW_IRGEN_FIFO fifo, HW_IRGEN_PAINT symbol, uint16_t duration)
{
    /* reg[20] = paint message (0), reg[14] = symbol, reg[13:0] = duration */
    uint16_t value = (!!symbol << 14) | (duration & 0x3FFF);
    if (fifo == HW_IRGEN_FIFO_REPEAT) {
        IRC->REPEAT_FIFO = value;
    } else {
        IRC->CODE_FIFO = value;
    }
}

/**
*\*\name    hw_irgen_flush_fifo.
*\*\fun     Flush FIFO.
*\*\param   fifo :
*\*\          - FIFO to be flushed
*\*\return  none
**/
static __INLINE void hw_irgen_flush_fifo(HW_IRGEN_FIFO fifo)
{
    if (fifo == HW_IRGEN_FIFO_REPEAT) {
        IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_REP_FIFO_RESET_Msk)) | \
            ((IR_CTRL_REP_FIFO_RESET_Msk) & ((1) << (IR_CTRL_REP_FIFO_RESET_Pos))));
    } else {
        IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_CODE_FIFO_RESET_Msk)) | \
            ((IR_CTRL_CODE_FIFO_RESET_Msk) & ((1) << (IR_CTRL_CODE_FIFO_RESET_Pos))));
    }
}

/** IRC output and TX control **/

/**
*\*\name    hw_irgen_set_output_type.
*\*\fun     Set output type (normal or inverted).
*\*\param   output :
*\*\          - output type
*\*\return  none
**/
static __INLINE void hw_irgen_set_output_type(HW_IRGEN_OUTPUT output)
{
	IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_INVERT_OUTPUT_Msk)) | \
	                ((IR_CTRL_INVERT_OUTPUT_Msk) & ((output) << (IR_CTRL_INVERT_OUTPUT_Pos))));
}

/**
*\*\name    hw_irgen_start.
*\*\fun     Start IR data transmission.
*\*\param   none
*\*\return  none
**/
static __INLINE void hw_irgen_start(void)
{
	IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_TX_START_Msk)) | \
	                ((IR_CTRL_TX_START_Msk) & ((1) << (IR_CTRL_TX_START_Pos))));

}

/**
*\*\name    hw_irgen_stop.
*\*\fun     Stop IR data transmission.
*\*\        When called after transmission is started, this will also flush code FIFO.
*\*\param   none
*\*\return  none
**/
static __INLINE void hw_irgen_stop(void)
{
	IRC->CTRL = ((IRC->CTRL & ~(IR_CTRL_TX_START_Msk)) | \
	                ((IR_CTRL_TX_START_Msk) & ((0) << (IR_CTRL_TX_START_Pos))));
}

/** IRC status and utility **/

/**
*\*\name    hw_irgen_is_busy.
*\*\fun     Check if IR generator is busy.
*\*\param   none
*\*\return  true if IR generator is busy, false otherwise
**/
static __INLINE bool hw_irgen_is_busy(void)
{
	return ((IRC->STATUS) & (IR_STATUS_BUSY));
}

/**
*\*\name    hw_irgen_get_fifo_level.
*\*\fun     Get number of messages in FIFO.
*\*\param   fifo :
*\*\          - FIFO
*\*\return  number of messages
**/
static __INLINE uint16_t hw_irgen_get_fifo_level(HW_IRGEN_FIFO fifo)
{
    if (fifo == HW_IRGEN_FIFO_REPEAT) {
        return (((IRC->STATUS) & (IR_STATUS_REP_FIFO_WORDS_Msk)) >> (IR_STATUS_REP_FIFO_WORDS_Pos));
    } else {
        return (((IRC->STATUS) & (IR_STATUS_CODE_FIFO_WORDS_Msk)) >> (IR_STATUS_CODE_FIFO_WORDS_Pos));
    }
}

/**
*\*\name    hw_irgen_reverse_bit_order.
*\*\fun     Reverse bit order.
*\*\        Helper function to reverse order of bits in message.
*\*\param   val :
*\*\          - message value
*\*\param   bits :
*\*\          - number of valid bits in val
*\*\return  val with bit order reversed
**/
static __INLINE uint16_t hw_irgen_reverse_bit_order(uint16_t val, uint8_t bits)
{
    uint16_t ret = 0;

    while (bits--) {
        ret = (ret << 1) | (val & 1);
        val >>= 1;
    }

    return ret;
}

}
