#ifndef __N32WB033_QFLASH_H__
#define __N32WB033_QFLASH_H__

#include "n32wb033.h"

#ifdef __cplusplus
extern "C" {
#endif

/** Flash operation return codes **/
typedef enum {
    FlashOperationSuccess,    /* Operation completed successfully            */
    FlashWriteRegFailed,      /* Failed to write status/configuration register */
    FlashTimeOut,             /* Operation timed out                         */
    FlashIsBusy,              /* Flash is busy (WIP=1)                       */
    FlashQuadNotEnable,       /* Quad mode not enabled                       */
    FlashAddressInvalid,      /* Address out of range                        */
} ReturnMsg;


/** QFlash_Exported_Functions **/

/** Init **/
void flash_init(void);

/** Erase / Program / Read **/
ReturnMsg flash_erase(uint32_t flash_address);
ReturnMsg flash_program(uint32_t flash_address, const uint8_t *source_address, uint32_t byte_length);
ReturnMsg flash_read(uint32_t flash_address, uint8_t *target_address, uint32_t byte_length);

/** OTP **/
ReturnMsg flash_otp_read(uint8_t security_reg_num, uint8_t* buffer, uint32_t buffer_size);


#ifdef __cplusplus
}
#endif

#endif /* __N32WB033_QFLASH_H__ */
