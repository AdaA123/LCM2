#ifndef __N32WB033_IWDG_H__
#define __N32WB033_IWDG_H__

#ifdef __cplusplus
extern "C" {
#endif

#include "n32wb033.h"

typedef enum
{
    IWDG_PVU_FLAG  = IWDG_STS_PVU,
    IWDG_CRVU_FLAG = IWDG_STS_CRVU,
    IWDG_FRZF_FLAG = IWDG_STS_FRZF,
} IWDG_STATUS_FLAG;

/** KEY register bit mask **/
#define KEY_ReloadKey                      ((uint16_t)0xAAAAU)
#define KEY_EnableKey                      ((uint16_t)0xCCCCU)
#define IWDG_FREEZE                        ((uint16_t)0x4567U)
#define IWDG_UNFREEZE                      ((uint16_t)0x89ABU)

/** PREDIV and RELV register write permission **/
typedef enum
{
    IWDG_WRITE_ENABLE  = 0x5555U,
    IWDG_WRITE_DISABLE = 0x0000U
} IWDOG_WRITE_CONFIG;


#define IWDG_PRESCALER_DIV4    ((uint8_t)0x00U)
#define IWDG_PRESCALER_DIV8    (IWDG_PREDIV_PD_0)
#define IWDG_PRESCALER_DIV16   (IWDG_PREDIV_PD_1)
#define IWDG_PRESCALER_DIV32   (IWDG_PREDIV_PD_1 | IWDG_PREDIV_PD_0)
#define IWDG_PRESCALER_DIV64   (IWDG_PREDIV_PD_2)
#define IWDG_PRESCALER_DIV128  (IWDG_PREDIV_PD_2 | IWDG_PREDIV_PD_0)
#define IWDG_PRESCALER_DIV256  (IWDG_PREDIV_PD_2 | IWDG_PREDIV_PD_1 | IWDG_PREDIV_PD_0)


/** Configuration **/
void IWDG_WriteConfig(IWDOG_WRITE_CONFIG IWDG_WriteAccess);
void IWDG_SetPrescalerDiv(uint8_t IWDG_Prescaler);
void IWDG_CntReload(uint16_t Reload);
void IWDG_ReloadKey(void);

/** Enable and disable **/
void IWDG_Enable(void);
void IWDG_Freeze_Enable(FunctionalState Cmd);

/** Flag management **/
FlagStatus IWDG_GetStatus(IWDG_STATUS_FLAG IWDG_FLAG);

#ifdef __cplusplus
}
#endif

#endif /* __N32WB033_IWDG_H__ */
