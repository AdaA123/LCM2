/**
*\*\name    TIM_InitTimeBase
*\*\fun     Initializes the TIMx time base unit.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\          - BTIM1
*\*\param   TIM_TimeBaseInitParam: refer to the definition of TIM_TimeBaseInitType
*\*\          - Prescaler
*\*\            * 0x0000 ~ 0xFFFF
*\*\          - CounterMode
*\*\            * TIM_CNT_MODE_UP
*\*\            * TIM_CNT_MODE_DOWN
*\*\            * TIM_CNT_MODE_CENTER_ALIGN1
*\*\            * TIM_CNT_MODE_CENTER_ALIGN2
*\*\            * TIM_CNT_MODE_CENTER_ALIGN3
*\*\          - Period
*\*\            * 0x00000000 ~ 0xFFFFFFFF
*\*\          - ClkDiv
*\*\            * TIM_CLK_DIV1
*\*\            * TIM_CLK_DIV2
*\*\            * TIM_CLK_DIV4
*\*\          - RepetCnt
*\*\            * 0x00 ~ 0xFF
*\*\return  none
**/
void TIM_InitTimeBase(TIM_Module* TIMx, const TIM_TimeBaseInitType* TIM_TimeBaseInitParam)
{
    uint32_t tmpctrl1;
    tmpctrl1 = TIMx->CTRL1;  
    
    if ((TIMx == ATIM1) || (TIMx == GTIM1)) 
    {
        /* Select the Counter Mode */
        tmpctrl1 &= (uint32_t)(~(TIM_CTRL1_DIR | TIM_CTRL1_CAMSEL));
        tmpctrl1 |= (uint32_t)TIM_TimeBaseInitParam->CounterMode;
    }
    else
    {
        /* No process */
    }
    if (TIMx == ATIM1) 
    {
        /* Set the clock division */
        tmpctrl1 &= (uint32_t)(~TIM_CTRL1_CLKD);
        tmpctrl1 |= (uint32_t)TIM_TimeBaseInitParam->ClkDiv;
    }   
    else
    {
        /* No process */
    }
    TIMx->CTRL1 = tmpctrl1;    
    
    /* Set the Autoreload value */
    TIMx->AR = TIM_TimeBaseInitParam->Period ;

    /* Set the Prescaler value */
    TIMx->PSC = TIM_TimeBaseInitParam->Prescaler;    
    
    if (TIMx == ATIM1) 
    {
        /* Set the Repetition Counter value */
        TIMx->REPCNT = TIM_TimeBaseInitParam->RepetCnt;
    }
    else
    {
        /* No process */
    }

    /* Generate an update event to reload the Prescaler and the Repetition counter
     values immediately */
    TIMx->EVTGEN = TIM_PSC_RELOAD_MODE_IMMEDIATE;
}

/**
*\*\name    TIM_InitOc1
*\*\fun     Initializes the TIMx Channel 1.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\param   TIM_OCInitParam: refer to the definition of OCInitType
*\*\          - OCMode
*\*\            * TIM_OCMODE_TIMING
*\*\            * TIM_OCMODE_ACTIVE
*\*\            * TIM_OCMODE_INACTIVE
*\*\            * TIM_OCMODE_TOGGLE
*\*\            * TIM_FORCED_ACTION_INACTIVE
*\*\            * TIM_FORCED_ACTION_ACTIVE
*\*\            * TIM_OCMODE_PWM1
*\*\            * TIM_OCMODE_PWM2
*\*\          - OutputState
*\*\            * TIM_OUTPUT_STATE_DISABLE
*\*\            * TIM_OUTPUT_STATE_ENABLE
*\*\          - OutputNState
*\*\            * TIM_OUTPUT_NSTATE_DISABLE
*\*\            * TIM_OUTPUT_NSTATE_ENABLE
*\*\          - Pulse
*\*\            * 0x00000000 ~ 0xFFFFFFFF
*\*\          - OCPolarity
*\*\            * TIM_OC_POLARITY_HIGH
*\*\            * TIM_OC_POLARITY_LOW
*\*\          - OCNPolarity
*\*\            * TIM_OCN_POLARITY_HIGH
*\*\            * TIM_OCN_POLARITY_LOW
*\*\          - OCIdleState
*\*\            * TIM_OC_IDLE_STATE_SET
*\*\            * TIM_OC_IDLE_STATE_RESET
*\*\          - OCNIdleState
*\*\            * TIM_OCN_IDLE_STATE_SET
*\*\            * TIM_OCN_IDLE_STATE_RESET
*\*\return  none
**/
void TIM_InitOc1(TIM_Module* TIMx, const OCInitType* TIM_OCInitParam)
{
    uint32_t tmpccmodx;
    uint32_t tmpccen, tmpctrl2;

    /* Disable the Channel 1: Reset the CC1EN Bit */
    TIMx->CCEN &= (uint32_t)(~TIM_CCEN_CC1EN);    
    /* Get the TIMx_CCEN register value */
    tmpccen = TIMx->CCEN;    
    /* Get the TIMx_CTRL2 register value */
    tmpctrl2 =  TIMx->CTRL2;
    /* Get the TIMx_CCMOD1 register value */
    tmpccmodx = TIMx->CCMOD1;

    /* Reset the Output Compare Mode Bits */
    tmpccmodx &= (uint32_t)(~TIM_CCMOD1_OC1MD); 
    tmpccmodx &= (uint32_t)(~TIM_CCMOD1_CC1SEL);    
    /* Select the Output Compare Mode */
    tmpccmodx |= (uint32_t)(TIM_OCInitParam->OCMode);

    /* Reset the Output Polarity level */
    tmpccen &= (uint32_t)(~TIM_CCEN_CC1P);    
    /* Set the Output Compare Polarity */
    tmpccen |= (uint32_t)(TIM_OCInitParam->OCPolarity);
    /* Set the Output State */
    tmpccen |= (uint32_t)(TIM_OCInitParam->OutputState);
    
    if (TIMx == ATIM1) 
    {
        /* Reset the Output N Polarity level */
        tmpccen &= (uint32_t)(~TIM_CCEN_CC1NP);
        /* Set the Output N Polarity */
        tmpccen |= (uint32_t)(TIM_OCInitParam->OCNPolarity);
        /* Reset the Output N State */
        tmpccen &= (uint32_t)(~TIM_CCEN_CC1NEN);    
        /* Set the Output N State */
        tmpccen |= (uint32_t)(TIM_OCInitParam->OutputNState);

        /* Reset the Output Compare and Output Compare N IDLE State */
        tmpctrl2 &= (uint32_t)(~TIM_CTRL2_OI1);
        tmpctrl2 &= (uint32_t)(~TIM_CTRL2_OI1N);
        /* Set the Output Idle state */
        tmpctrl2 |= (uint32_t)(TIM_OCInitParam->OCIdleState);
        /* Set the Output N Idle state */
        tmpctrl2 |= (uint32_t)(TIM_OCInitParam->OCNIdleState);
    }    
    
    /* Write to TIMx_CTRL2 */
    TIMx->CTRL2 = tmpctrl2;

    /* Write to TIMx_CCMOD1 */
    TIMx->CCMOD1 = tmpccmodx;

    /* Set the Capture Compare Register value */
    TIMx->CCDAT1 = TIM_OCInitParam->Pulse; 

    /* Write to TIMx_CCEN */
    TIMx->CCEN = tmpccen;    
}

/**
*\*\name    TIM_InitOc2
*\*\fun     Initializes the TIMx Channel 2.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\param   TIM_OCInitParam: refer to the definition of OCInitType

*\*\            * TIM_OUTPUT_STATE_DISABLE
*\*\            * TIM_OUTPUT_STATE_ENABLE
*\*\          - OutputNState
*\*\            * TIM_OUTPUT_NSTATE_DISABLE
*\*\            * TIM_OUTPUT_NSTATE_ENABLE
*\*\          - Pulse
*\*\            * 0x00000000 ~ 0xFFFFFFFF
*\*\          - OCPolarity
*\*\            * TIM_OC_POLARITY_HIGH
*\*\            * TIM_OC_POLARITY_LOW
*\*\          - OCNPolarity
*\*\            * TIM_OCN_POLARITY_HIGH
*\*\            * TIM_OCN_POLARITY_LOW
*\*\          - OCIdleState
*\*\            * TIM_OC_IDLE_STATE_SET
*\*\            * TIM_OC_IDLE_STATE_RESET
*\*\          - OCNIdleState
*\*\            * TIM_OCN_IDLE_STATE_SET
*\*\            * TIM_OCN_IDLE_STATE_RESET
*\*\return  none
**/
void TIM_InitOc2(TIM_Module* TIMx, const OCInitType* TIM_OCInitParam)
{
    uint32_t tmpccmodx ;
    uint32_t tmpccen , tmpctrl2;

    /* Disable the Channel 2: Reset the CC2EN Bit */
    TIMx->CCEN &= (uint32_t)(~TIM_CCEN_CC2EN);

    /* Get the TIMx_CCEN register value */  
    tmpccen = TIMx->CCEN;
    /* Get the TIMx_CTRL2 register value */
    tmpctrl2 =  TIMx->CTRL2;

    /* Get the TIMx_CCMOD1 register value */
    tmpccmodx = TIMx->CCMOD1;

    /* Reset the Output Compare mode and Capture/Compare selection Bits */
    tmpccmodx &= (uint32_t)(~TIM_CCMOD1_OC2MD);
    tmpccmodx &= (uint32_t)(~TIM_CCMOD1_CC2SEL);

    /* Select the Output Compare Mode */
    tmpccmodx |= (uint32_t)(TIM_OCInitParam->OCMode << (uint32_t)8U);      
    

    /* Reset the Output Polarity level */
    tmpccen &= (uint32_t)(~TIM_CCEN_CC2P);    
    /* Set the Output Compare Polarity */
    tmpccen |= (uint32_t)(TIM_OCInitParam->OCPolarity << (uint32_t)4U);  
    /* Set the Output State */
    tmpccen |= (uint32_t)(TIM_OCInitParam->OutputState << (uint32_t)4U);    
       
    if (TIMx == ATIM1)
    {
        /* Reset the Output N Polarity level */
        tmpccen &= (uint32_t)(~TIM_CCEN_CC2NP);
        /* Set the Output N Polarity */
        tmpccen |= (uint32_t)(TIM_OCInitParam->OCNPolarity << (uint32_t)4U);   
        /* Reset the Output N State */
        tmpccen &= (uint32_t)(~TIM_CCEN_CC2NEN);    
        /* Set the Output N State */
        tmpccen |= (uint32_t)(TIM_OCInitParam->OutputNState << (uint32_t)4U);   
        
        /* Reset the Output Compare and Output Compare N IDLE State */
        tmpctrl2 &= (uint32_t)(~TIM_CTRL2_OI2);
        tmpctrl2 &= (uint32_t)(~TIM_CTRL2_OI2N);
        /* Set the Output Idle state */
        tmpctrl2 |= (uint32_t)(TIM_OCInitParam->OCIdleState << (uint32_t)2U);
        /* Set the Output N Idle state */
        tmpctrl2 |= (uint32_t)(TIM_OCInitParam->OCNIdleState << (uint32_t)2U);        
    }     
    
    /* Write to TIMx_CTRL2 */
    TIMx->CTRL2 = tmpctrl2;

    /* Write to TIMx_CCMOD1 */
    TIMx->CCMOD1 = tmpccmodx;

    /* Set the Capture Compare Register value */
    TIMx->CCDAT2 = TIM_OCInitParam->Pulse;
 
    /* Write to TIMx_CCEN */
    TIMx->CCEN = tmpccen;    
}

/**
*\*\name    TIM_ICInit
*\*\fun     Initializes the TIM peripheral according to the specified
*\*\            parameters in the TIM_ICInitParam.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\param   TIM_ICInitParam: refer to the definition of TIM_ICInitType
*\*\          - Channel
*\*\            * TIM_CH_1
*\*\            * TIM_CH_2
*\*\            * TIM_CH_3
*\*\            * TIM_CH_4
*\*\          - ICPolarity
*\*\            * TIM_IC_POLARITY_RISING
*\*\            * TIM_IC_POLARITY_FALLING
*\*\          - ICSelection
*\*\            * TIM_IC_SELECTION_DIRECTTI
*\*\            * TIM_IC_SELECTION_INDIRECTTI
*\*\            * TIM_IC_SELECTION_TRC
*\*\          - ICPrescaler
*\*\            * TIM_IC_PSC_DIV1
*\*\            * TIM_IC_PSC_DIV2
*\*\            * TIM_IC_PSC_DIV4
*\*\            * TIM_IC_PSC_DIV8
*\*\          - ICFilter
*\*\            * ICFilter can be a number between 0x0 and 0xF
*\*\return  none
**/
void TIM_ICInit(TIM_Module* TIMx, const TIM_ICInitType* TIM_ICInitParam)
{
    if (TIM_ICInitParam->Channel == TIM_CH_1)
    {
        /* TI1 Configuration */
        ConfigTI1(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap1Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }
    else if (TIM_ICInitParam->Channel == TIM_CH_2)
    {
        /* TI2 Configuration */
        ConfigTI2(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap2Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }
    else if (TIM_ICInitParam->Channel == TIM_CH_3)
    {
        /* TI3 Configuration */
        ConfigTI3(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap3Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }    
    else if (TIM_ICInitParam->Channel == TIM_CH_4)
    {
        /* TI4 Configuration */
        ConfigTI4(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap4Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }
    else
    {
        /* No process */
    }    
}

*\*\            * TIM_IC_SELECTION_INDIRECTTI
*\*\          - ICPrescaler
*\*\            * TIM_IC_PSC_DIV1
*\*\            * TIM_IC_PSC_DIV2

*\*\            * TIM_IC_PSC_DIV8
*\*\          - ICFilter
*\*\            * ICFilter can be a number between 0x0 and 0xF
*\*\return  none
**/
void TIM_ConfigPwmIc(TIM_Module* TIMx, const TIM_ICInitType* TIM_ICInitParam)
{
    uint32_t icoppositepolarity  ;
    uint32_t icoppositeselection ;

    /* Select the Opposite Input Polarity */
    if (TIM_ICInitParam->ICPolarity == TIM_IC_POLARITY_RISING)
    {
        icoppositepolarity = TIM_IC_POLARITY_FALLING;
    }
    else
    {
        icoppositepolarity = TIM_IC_POLARITY_RISING;
    }
    /* Select the Opposite Input */
    if (TIM_ICInitParam->ICSelection == TIM_IC_SELECTION_DIRECTTI)
    {
        icoppositeselection = TIM_IC_SELECTION_INDIRECTTI;
    }
    else
    {
        icoppositeselection = TIM_IC_SELECTION_DIRECTTI;
    }
    if (TIM_ICInitParam->Channel == TIM_CH_1)
    {
        /* TI1 Configuration */
        ConfigTI1(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap1Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
        /* TI2 Configuration */
        ConfigTI2(TIMx, icoppositepolarity, icoppositeselection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap2Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }
    else
    {
        /* TI2 Configuration */
        ConfigTI2(TIMx, TIM_ICInitParam->ICPolarity, TIM_ICInitParam->ICSelection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap2Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
        /* TI1 Configuration */
        ConfigTI1(TIMx, icoppositepolarity, icoppositeselection, TIM_ICInitParam->ICFilter);
        /* Set the Input Capture Prescaler value */
        TIM_SetInCap1Prescaler(TIMx, TIM_ICInitParam->ICPrescaler);
    }
}

/**
*\*\name    TIM_ConfigBkdt
*\*\fun     Configures the: Break feature, dead time, Lock level, the OSSI,
*\*\            the OSSR State and the AOE(automatic output enable).
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\param   TIM_BDTRInitParam: refer to the definition of TIM_BDTRInitType
*\*\          - OSSRState
*\*\            * TIM_OSSR_STATE_ENABLE
*\*\            * TIM_OSSR_STATE_DISABLE
*\*\          - OSSIState
*\*\            * TIM_OSSI_STATE_ENABLE
*\*\            * TIM_OSSI_STATE_DISABLE
*\*\          - LOCKLevel
*\*\            * TIM_LOCK_LEVEL_OFF
*\*\            * TIM_LOCK_LEVEL_1
*\*\            * TIM_LOCK_LEVEL_2
*\*\            * TIM_LOCK_LEVEL_3
*\*\          - DeadTime
*\*\            * DeadTime can be a number between 0x00 and 0xFF
*\*\          - Break
*\*\            * TIM_BREAK_IN_ENABLE
*\*\            * TIM_BREAK_IN_DISABLE
*\*\          - BreakPolarity
*\*\            * TIM_BREAK_POLARITY_LOW
*\*\            * TIM_BREAK_POLARITY_HIGH
*\*\          - AutomaticOutput
*\*\            * TIM_AUTO_OUTPUT_ENABLE
*\*\            * TIM_AUTO_OUTPUT_DISABLE
*\*\return  none
**/
void TIM_ConfigBkdt(TIM_Module* TIMx, const TIM_BDTRInitType* TIM_BDTRInitParam)
{
    uint32_t tmpbkdt;
    
    /* Set the Lock level, the OSSR State, the OSSI State, 
           the dead time value and the Automatic Output Enable Bit */
    tmpbkdt = TIMx->BKDT;
    tmpbkdt &= (uint32_t)(~(TIM_BKDT_OSSR | TIM_BKDT_OSSI | TIM_BKDT_LCKCFG | TIM_BKDT_DTGN | TIM_BKDT_AOEN));
    tmpbkdt |= (uint32_t)(TIM_BDTRInitParam->OSSRState | TIM_BDTRInitParam->OSSIState | TIM_BDTRInitParam->LOCKLevel |\
                          TIM_BDTRInitParam->DeadTime | TIM_BDTRInitParam->AutomaticOutput);

    /* Set the Break enable Bit and the Ploarity */
    tmpbkdt &= (uint32_t)(~(TIM_BKDT_BKEN | TIM_BKDT_BKP ));        
    tmpbkdt |= (uint32_t)(TIM_BDTRInitParam->Break | TIM_BDTRInitParam->BreakPolarity);
    TIMx->BKDT = tmpbkdt;    
}

/**
*\*\name    TIM_BreakFiltConfig
*\*\fun     Config the filter of break input.
*\*\param   TIMx:
*\*\          - ATIM1
*\*\param   TIM_FiltInitParam: 
*\*\          - refer to the definition of TIM_FiltInitType
*\*\return  none
**/
void TIM_BreakFiltConfig(TIM_Module* TIMx, const TIM_FiltInitType* TIM_FiltInitParam)
{
    uint32_t tempbkfr;

    tempbkfr = TIMx->BKFR;
    
    tempbkfr &= (uint32_t)(((uint32_t) ~((uint32_t)TIM_BKFR_THRESH)) & 
                             ((uint32_t) ~((uint32_t)TIM_BKFR_WSIZE)));

    tempbkfr |= (uint32_t)((TIM_FiltInitParam->ThreshHold << (uint32_t)0x18U) | 
                             (TIM_FiltInitParam->WindowSize << (uint32_t)0x11U));        
    TIMx->BKFR = tempbkfr;

    if (TIMx == ATIM1) 
    {
        /* Set the filt*/
        RCC_ConfigATim1Filter(TIM_FiltInitParam->Prescaler);
    }
    else
    {
        /* No process */
    }
}

/**
*\*\name    TIM_BreakInputSourceEnable
*\*\fun     Enable or disable the break input source , and set the polarity of the break input source.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\param   Source
*\*\          - TIM_BREAK_LOCKUP
*\*\          - TIM_BREAK_IOM1
*\*\          - TIM_BREAK_IOM2
*\*\          - TIM_BREAK_IOM3
*\*\          - TIM_BREAK_IOM4
*\*\param   Polarity
*\*\          - TIM_BREAK_SOURCE_POLARITY_INVERT
*\*\          - TIM_BREAK_SOURCE_POLARITY_NONINVERT
*\*\param   Cmd
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
*\*\note Only a system reset can reset TIM_BREAK_LOCKUP after they are enabled.
**/
void TIM_BreakInputSourceEnable(TIM_Module* TIMx, uint32_t Source, uint32_t Polarity, FunctionalState Cmd)
{
    uint32_t tempctrl1;
    uint32_t tempaf1;
    tempctrl1 = TIMx->CTRL1;
    tempaf1 = TIMx->AF1;
    
    if (Cmd != DISABLE)
    {
        if (IS_SYS_BREAK(Source))
        {
            tempctrl1 |= (uint32_t)(Source);        
        }
        else
        {
            /* Enable break input source */ 
            tempaf1 |= (uint32_t)(Source);   
        } 

    }
    else
    {
        if (IS_SYS_BREAK(Source))
        {
            /* Only a system reset can reset TIM_BREAK_LOCKUP after they are enabled. */        
        }
        else
        {
            /* Disable break input source */             
            tempaf1 &= (uint32_t)(~Source);                
        }        
    }
    if(Polarity==TIM_BREAK_SOURCE_POLARITY_INVERT)
    {
        /* Set break input source polarity */
        if(Source==TIM_BREAK_IOM1)
        {
            tempaf1 |= (uint32_t)(TIM_BREAK_IOM1P); 
        }
        else if(Source==TIM_BREAK_IOM2)
        {
            tempaf1 |= (uint32_t)(TIM_BREAK_IOM2P); 
        }  
        else if(Source==TIM_BREAK_IOM3)
        {
            tempaf1 |= (uint32_t)(TIM_BREAK_IOM3P); 
        } 
        else if(Source==TIM_BREAK_IOM4)
        {
            tempaf1 |= (uint32_t)(TIM_BREAK_IOM4P); 
        } 
        else
        {
            /* No process */
        }
    }
    else
    {
        /* Set break input source polarity */
        if(Source==TIM_BREAK_IOM1)
        {
            tempaf1 &= (uint32_t)(~TIM_BREAK_IOM1P); 
        }
        else if(Source==TIM_BREAK_IOM2)
        {
            tempaf1 &= (uint32_t)(~TIM_BREAK_IOM2P); 
        }  
        else if(Source==TIM_BREAK_IOM3)
        {
            tempaf1 &= (uint32_t)(~TIM_BREAK_IOM3P); 
        } 
        else if(Source==TIM_BREAK_IOM4)
        {
            tempaf1 &= (uint32_t)(~TIM_BREAK_IOM4P); 
        } 
        else
        {
            /* No process */
        }
    }
    TIMx->CTRL1 = tempctrl1;
    TIMx->AF1 = tempaf1;    
}

*\*\          - TIM_EXT_TRG_PSC_OFF
*\*\          - TIM_EXT_TRG_PSC_DIV2
*\*\          - TIM_EXT_TRG_PSC_DIV4
*\*\          - TIM_EXT_TRG_PSC_DIV8
*\*\param   TIM_ExtTRGPolarity
*\*\          - TIM_EXT_TRIG_POLARITY_INVERTED
*\*\          - TIM_EXT_TRIG_POLARITY_NONINVERTED
*\*\param   ExtTRGFilter
*\*\          - ExtTRGFilter can be a number between 0x0 and 0xF
*\*\return  none
**/
void TIM_ConfigExtClkMode1(TIM_Module* TIMx, uint32_t TIM_ExtTRGPrescaler, uint32_t TIM_ExtTRGPolarity, uint32_t ExtTRGFilter)
{
    uint32_t tmpsmctrl ;
    /* Configure the ETR Clock source */
    TIM_ConfigExtTrig(TIMx, TIM_ExtTRGPrescaler, TIM_ExtTRGPolarity, ExtTRGFilter);

    /* Get the TIMx SMCTRL register value */
    tmpsmctrl = TIMx->SMCTRL;
    /* Reset the SMS Bits */
    tmpsmctrl &= (uint32_t)(~((uint32_t)TIM_SMCTRL_SMSEL));
    /* Select the External clock mode1 */
    tmpsmctrl |= (uint32_t)TIM_SLAVE_MODE_EXT1;
    /* Select the Trigger selection : ETRF */
    tmpsmctrl &= (uint32_t)(~((uint32_t)TIM_SMCTRL_TSEL));
    tmpsmctrl |= (uint32_t)TIM_TRIG_SEL_ETRF;
    /* Write to TIMx SMCTRL */
    TIMx->SMCTRL = tmpsmctrl;
}

/**
*\*\name    TIM_ConfigEncoderInterface
*\*\fun     Configures the TIMx Encoder Interface.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\param   TIM_EncoderMode
*\*\          - TIM_ENCODE_QUA_MODE_TI1
*\*\          - TIM_ENCODE_QUA_MODE_TI2
*\*\          - TIM_ENCODE_QUA_MODE_TI12
*\*\param   TIM_IC1Polarity
*\*\          - TIM_IC_POLARITY_FALLING
*\*\          - TIM_IC_POLARITY_RISING
*\*\param   TIM_IC2Polarity
*\*\          - TIM_IC_POLARITY_FALLING
*\*\          - TIM_IC_POLARITY_RISING
*\*\return  none
**/
void TIM_ConfigEncoderInterface(TIM_Module* TIMx, uint32_t TIM_EncoderMode, uint32_t TIM_IC1Polarity, uint32_t TIM_IC2Polarity)
{
    uint32_t tmpsmctrl  ;
    uint32_t tmpccmodx ;
    uint32_t tmpccen  ;
    /* Get the TIMx SMCTRL register value */
    tmpsmctrl = TIMx->SMCTRL;

    /* Get the TIMx CCMOD1 register value */
    tmpccmodx = TIMx->CCMOD1;

    /* Get the TIMx CCEN register value */
    tmpccen = TIMx->CCEN;

    /* Set the encoder Mode */
    tmpsmctrl &= (uint32_t)(~((uint32_t)TIM_SMCTRL_SMSEL));
    tmpsmctrl |= (uint32_t)TIM_EncoderMode;

    /* Select the Capture Compare 1 and the Capture Compare 2 as input */
    tmpccmodx &= (uint32_t)((~TIM_CCMOD1_CC1SEL) & (~TIM_CCMOD1_CC2SEL));
    tmpccmodx |= (uint32_t)(TIM_CCMOD1_CC1SEL_0 | TIM_CCMOD1_CC2SEL_0);

    /* Set the TI1 and the TI2 Polarities */
    tmpccen &= (uint32_t)(((uint32_t) ~((uint32_t)TIM_CCEN_CC1P)) & ((uint32_t) ~((uint32_t)TIM_CCEN_CC2P)));
    tmpccen |= (uint32_t)(TIM_IC1Polarity | (uint32_t)(TIM_IC2Polarity << (uint32_t)4U));

    /* Write to TIMx SMCTRL */
    TIMx->SMCTRL = tmpsmctrl;
    /* Write to TIMx CCMOD1 */
    TIMx->CCMOD1 = tmpccmodx;
    /* Write to TIMx CCEN */
    TIMx->CCEN = tmpccen;
}

/**
*\*\name    TIM_SelectOcMode
*\*\fun     Selects the TIM Output Compare Mode.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\param   Channel
*\*\          - TIM_CH_1
*\*\          - TIM_CH_2
*\*\          - TIM_CH_3
*\*\          - TIM_CH_4
*\*\param   OcMode
*\*\          - TIM_OCMODE_TIMING
*\*\          - TIM_OCMODE_ACTIVE
*\*\          - TIM_OCMODE_INACTIVE
*\*\          - TIM_OCMODE_TOGGLE
*\*\          - TIM_OCMODE_PWM1
*\*\          - TIM_OCMODE_PWM2
*\*\          - TIM_FORCED_ACTION_ACTIVE
*\*\          - TIM_FORCED_ACTION_INACTIVE
*\*\return  none
**/
void TIM_SelectOcMode(TIM_Module* TIMx, uint32_t Channel, uint32_t OcMode)
{
    uint32_t tmp  ;
    uint32_t tmp1 ;

    tmp = (uint32_t)(uintptr_t)TIMx;
    tmp += CAPCMPMOD_OFFSET;

    tmp1 = CAPCMPEN_CCE_SET << Channel;

    /* Disable the Channel: Reset the CCxE Bit */
    TIMx->CCEN &= (uint32_t)(~tmp1);

    if ((Channel == TIM_CH_1) || (Channel == TIM_CH_3))
    {
        tmp += (uint32_t)(Channel >> 1);

        /* Reset the OCxMD bits in the CCMRx register */
        *((__IO uint32_t*)(uintptr_t)tmp) &= (uint32_t) (~TIM_CCMOD1_OC1MD);

        /* Configure the OCxMD bits in the CCMRx register */
        *((__IO uint32_t*)(uintptr_t)tmp) |= (uint32_t)OcMode;
    }
    else if ((Channel == TIM_CH_2) || (Channel == TIM_CH_4))
    {
        tmp += (uint32_t)((Channel - 4U) >> 1);

        /* Reset the OCxMD bits in the CCMRx register */
        *((__IO uint32_t*)(uintptr_t)tmp) &= (uint32_t)(~TIM_CCMOD1_OC2MD);

        /* Configure the OCxMD bits in the CCMRx register */
        *((__IO uint32_t*)(uintptr_t)tmp) |= (uint32_t)(OcMode << (uint32_t)8U);
    }
    else
    {
        /* No process */
    }
}

/**
*\*\name    ConfigTI1
*\*\fun     Configure the TI1 as Input.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\param   ICPolarity
*\*\          - TIM_IC_POLARITY_RISING
*\*\          - TIM_IC_POLARITY_FALLING
*\*\param   ICSelection
*\*\          - TIM_IC_SELECTION_DIRECTTI
*\*\          - TIM_IC_SELECTION_INDIRECTTI
*\*\          - TIM_IC_SELECTION_TRC
*\*\param   ICFilter
*\*\          - ICFilter can be a number between 0x0 and 0xF
*\*\return  none
**/
void ConfigTI1(TIM_Module* TIMx, uint32_t ICPolarity, uint32_t ICSelection, uint32_t ICFilter)
{
    uint32_t tmpccmodx ;
    uint32_t tmpccen  ;
    
    /* Disable the Channel 1: Reset the CC1E Bit */
    TIMx->CCEN &= (uint32_t)(~((uint32_t)TIM_CCEN_CC1EN));    

    /* Get the TIMx_CCMOD1 and TIMx_CCEN register value */    
    tmpccmodx = TIMx->CCMOD1;
    tmpccen  = TIMx->CCEN;    
    
    /* Select the Input and set the filter */
    tmpccmodx &= (uint32_t)(((uint32_t) ~((uint32_t)TIM_CCMOD1_CC1SEL)) & ((uint32_t) ~((uint32_t)TIM_CCMOD1_IC1F)));
    tmpccmodx |= (uint32_t)(ICSelection | (uint32_t)(ICFilter << (uint32_t)4U));
    
    /* Select the Polarity and set the CC1E Bit */
    tmpccen &= (uint32_t)(~((uint32_t)(TIM_CCEN_CC1P)));
    tmpccen |= (uint32_t)(ICPolarity | (uint32_t)TIM_CCEN_CC1EN);   
    
    /* Write to TIMx CCMOD1 and CCEN registers */
    TIMx->CCMOD1 = tmpccmodx;
    TIMx->CCEN   = tmpccen;    
}

*\*\          - TIM_EXT_TRG_PSC_OFF

*\*\          - TIM_EXT_TRIG_POLARITY_NONINVERTED
*\*\param   ExtTRGFilter

*\*\return  none
**/

    tmpsmctrl |= (uint32_t)(TIM_ExtTRGPrescaler |(TIM_ExtTRGPolarity | (ExtTRGFilter << (uint32_t)12U)));
    /* Write to TIMx SMCTRL */
    TIMx->SMCTRL = tmpsmctrl;

/**
*\*\name    TIM_GetIntStatus
*\*\fun     Checks whether the TIM interrupt has occurred or not.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - ATIM1
*\*\          - GTIM1
*\*\          - BTIM1
*\*\param   TIM_IT :
*\*\          - TIM_INT_UPDATE
*\*\          - TIM_INT_CC1
*\*\          - TIM_INT_CC2
*\*\          - TIM_INT_CC3
*\*\          - TIM_INT_CC4
*\*\          - TIM_INT_C3LDCNT
*\*\          - TIM_INT_COM
*\*\          - TIM_INT_TRIG
*\*\          - TIM_INT_BREAK
*\*\return  The new state of the TIM_IT(SET or RESET).
**/
INTStatus TIM_GetIntStatus(const TIM_Module* TIMx, uint32_t TIM_IT)
{
    INTStatus bitstatus ;
    uint32_t itstatus = 0x0U, itenable;
    if(TIM_IT == TIM_INT_UPDATE)
    {
        itstatus = TIMx->STS & TIM_FLAG_UPDATE;
    }
    else if(TIM_IT == TIM_INT_CC1)
    {
        itstatus = TIMx->STS & TIM_FLAG_CC1;
    }  
    else if(TIM_IT == TIM_INT_CC2)
    {
        itstatus = TIMx->STS & TIM_FLAG_CC2;
    }  
    else if(TIM_IT == TIM_INT_CC3)
    {
        itstatus = TIMx->STS & TIM_FLAG_CC3;
    }  
    else if(TIM_IT == TIM_INT_CC4)
    {
        itstatus = TIMx->STS & TIM_FLAG_CC4;
    }  
    else if(TIM_IT == TIM_INT_C3LDCNT)
    {
        itstatus = TIMx->STS & TIM_FLAG_C3LDCNT;
    }
    else if(TIM_IT == TIM_INT_COM)
    {
        itstatus = TIMx->STS & TIM_FLAG_COM;
    }  
    else if(TIM_IT == TIM_INT_TRIG)
    {
        itstatus = TIMx->STS & TIM_FLAG_TRIG;
    } 
    else if(TIM_IT == TIM_INT_BREAK)
    {
        itstatus = TIMx->STS & TIM_FLAG_BREAK;
    }  
    else
    {
        /* No process */
    } 

    itenable = TIMx->DINTEN & TIM_IT;
    if ((itstatus != (uint32_t)RESET) && (itenable != (uint32_t)RESET))
    {
        bitstatus = SET;
    }
    else
    {
        bitstatus = RESET;
    }
    return bitstatus;
}

/**

*\*\param   TIMx:
*\*\          - GTIM1 
*\*\param   TIM_FiltInitParam:

*\*\return  none
**/

{
    uint32_t temp_chxfilt = 0U;
    
    temp_chxfilt &= (uint32_t)(((uint32_t) ~((uint32_t)TIM_C1FILT_THRESH)) & 
                             ((uint32_t) ~((uint32_t)TIM_C1FILT_WSIZE)));

    temp_chxfilt |= (uint32_t)((TIM_FiltInitParam->ThreshHold << (uint32_t)0x18U) | 
                             (TIM_FiltInitParam->WindowSize << (uint32_t)0x11U));
 

    TIMx->C1FILT = temp_chxfilt;
    RCC_ConfigGTim1Filter(TIM_FiltInitParam->Prescaler);
}

/**
*\*\name    TIM_CH3LoadLvrToCNTPolarity
*\*\fun     Config CH3 Polarity to trigger ENCLVR value loading into the CNT.
*\*\param   TIMx (The input parameters must be the following values):
*\*\          - GTIM1
*\*\param   Polarity :
*\*\          - TIM_CH3_POLARITY_HIGH
*\*\          - TIM_CH3_POLARITY_LOW
*\*\          - TIM_CH3_POLARITY_RISING
*\*\          - TIM_CH3_POLARITY_FALLING
*\*\return  none
*\*\note  only for TIM4
**/
void TIM_CH3LoadLvrToCNTPolarity(TIM_Module* TIMx, uint32_t Polarity)
{
    uint32_t tmpencmctrl ;
    tmpencmctrl = TIMx->ENCMCTRL;
    /* Config the C3LDCNTSEL Bit */
    tmpencmctrl &= (uint32_t)(~((uint32_t)TIM_ENCMCTRL_C3LDCNTSEL));
    tmpencmctrl |= (uint32_t)Polarity;
    /* Write to TIMx ENCMCTRL register */
    TIMx->ENCMCTRL = tmpencmctrl;

}
