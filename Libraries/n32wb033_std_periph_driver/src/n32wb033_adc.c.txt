/* bit data handler */
#define BIT_JUDGE_AND_HANDLER(IS_ENABLE, BIT_MASK)    (((IS_ENABLE) != DISABLE ) ? (BIT_MASK) : (0x00000000U))

#define CTRL1_CLR_MASK          ((uint32_t)0x00000100U)
#define CTRL2_CLR_MASK          ((uint32_t)0x001E0802U)
#define CTRL3_CLR_MASK          ((uint32_t)0x00000003U)
#define RSEQ1_CLR_MASK          ((uint32_t)0x00F00000U)

/** ADC Driving Functions Declaration **/

/**
*\*\name    ADC_DeInit.
*\*\fun     Reset the ADC registers.
*\*\return  none
**/
void ADC_DeInit(void)
{
    RCC_EnableAHBPeriphReset(RCC_AHB_PERIPH_ADC);
}


/**
*\*\name    ADC_Init.
*\*\fun     Initializes the ADC peripheral according to the specified parameters in the ADC_InitParam.
*\*\param   ADC_InitParam :
*\*\          - MultiChEn
*\*\             - ENABLE
*\*\             - DISABLE
*\*\          - ContinueConvEn
*\*\             - ENABLE
*\*\             - DISABLE
*\*\          - ExtTrigSelect
*\*\            - ADC_EXT_TRIG_REG_CONV_ATIM1_CC1
*\*\            - ADC_EXT_TRIG_REG_CONV_ATIM1_CC2
*\*\            - ADC_EXT_TRIG_REG_CONV_ATIM1_CC3
*\*\            - ADC_EXT_TRIG_REG_CONV_GTIM1_CC1
*\*\            - ADC_EXT_TRIG_REG_CONV_GTIM1_CC2
*\*\            - ADC_EXT_TRIG_REG_CONV_GTIM1_TRGO
*\*\            - ADC_EXT_TRIG_REG_CONV_EXTI_LINE0_15_OR_ATIM1_TRGO
*\*\            - ADC_EXT_TRIG_REG_CONV_SOFTWARE
*\*\          - DatAlign
*\*\             - ADC_DAT_ALIGN_R
*\*\             - ADC_DAT_ALIGN_L
*\*\          - Resolution
*\*\             - ADC_DATA_RES_12BIT
*\*\             - ADC_DATA_RES_10BIT
*\*\             - ADC_DATA_RES_8BIT
*\*\             - ADC_DATA_RES_6BIT
*\*\          - ChsNumber: This parameter must be between 1 to 16.
*\*\return  none
**/
void ADC_Init(const ADC_InitType* ADC_InitParam)
{
    uint32_t tempreg1;
    uint8_t tempreg2;

    /*---------------------------- ADCx PCTRL Configuration -----------------*/
    /* Get the ADCx PCTRL value */
    tempreg1 = ADC->PCTRL;
    /* clear ADC_PCTRL register EXPEN RANGESEL VREFLDOEN bits */
    tempreg1 &= (~(ADC_PCTRL_EXPEN_MASK | ADC_PCTRL_RANGESEL_MASK | ADC_PCTRL_VREFLDOEN_MASK));
    /* Set EXPEN RANGESEL VREFLDOEN select bits */
    tempreg1 |= ADC_EXTERN_POWER_SUPPORT;
     /* Store the new register value */
    ADC->PCTRL = tempreg1;
    
    /*---------------------------- ADC CTRL1 Configuration -----------------*/
    /* Get the ADC CTRL1 value */
    tempreg1 = ADC->CTRL1;
    /* Clear SCANMD bits */
    tempreg1 &= (~CTRL1_CLR_MASK);
    /* Set SCANMD bit according to MultiChEn value */
    tempreg1 |= ((uint32_t)ADC_InitParam->MultiChEn << (uint32_t)8U);
    /* Write to ADC CTRL1 */
    ADC->CTRL1 = tempreg1;

    /*---------------------------- ADC CTRL2 Configuration -----------------*/
    /* Get the ADC CTRL2 value */
    tempreg1 = ADC->CTRL2;
    /* Clear CTU, ALIG, EXTRSEL and TRIG bits */
    tempreg1 &= (~CTRL2_CLR_MASK);
    /* Set ALIGN bit according to DatAlign value */
    /* Set EXTSEL bits according to ExtTrigSelect value */
    /* Set CTU bit according to ContinueConvEn value */
    tempreg1 |= (uint32_t)(ADC_InitParam->DatAlign | ADC_InitParam->ExtTrigSelect
                          | ((uint32_t)ADC_InitParam->ContinueConvEn << (uint32_t)1U));
    /* Write to ADC CTRL2 */
    ADC->CTRL2 = tempreg1;
    		/*---------------------------- ADCx CTRL3 Configuration -----------------*/
    /* Get the ADCx CTRL3 value */
    tempreg1 = ADC->CTRL3;
    /* Clear bits */
    tempreg1 &= (~(CTRL3_CLR_MASK | ADC_CTRL3_BUFF_EN));
    /* Configure ADCx resolution */
    /* Set RES bits according to Resolution value */
    tempreg1 |= ADC_InitParam->Resolution;
    /* Write to ADCx CTRL3 */
    ADC->CTRL3 = tempreg1;
    /*---------------------------- ADC RSEQ1 Configuration -----------------*/
    /* Get the ADC RSEQ1 value */
    tempreg1 = ADC->RSEQ1;
    /* Clear L bits */
    tempreg1 &= (~RSEQ1_CLR_MASK);
    /* Configure ADC: regular channel sequence length */
    /* Set LEN bits according to ChsNumber value */
    tempreg2 = (uint8_t)(ADC_InitParam->ChsNumber - 1U);
    tempreg1 |= (uint32_t)tempreg2 << 20U;
    /* Write to ADC RSEQ1 */
    ADC->RSEQ1 = tempreg1;
}


/**
*\*\name    ADC_InitStruct.
*\*\fun     Fills all ADC_InitParam member with default value.
*\*\param   ADC_InitParam :
*\*\          - MultiChEn
*\*\          - ContinueConvEn
*\*\          - ExtTrigSelect
*\*\          - DatAlign
*\*\          - ChsNumber
*\*\return  none
**/
void ADC_InitStruct(ADC_InitType* ADC_InitParam)
{
    /* Reset ADC init structure parameters values */
    /* initialize the MultiChEn member */
    ADC_InitParam->MultiChEn = DISABLE;
    /* Initialize the ContinueConvEn member */
    ADC_InitParam->ContinueConvEn = DISABLE;
    /* Initialize the ExtTrigSelect member */
    ADC_InitParam->ExtTrigSelect = ADC_EXT_TRIG_REG_CONV_SOFTWARE;
    /* Initialize the DatAlign member */
    ADC_InitParam->DatAlign = ADC_DAT_ALIGN_R;
    /* Initialize the ChsNumber member */
    ADC_InitParam->ChsNumber = 1U;
     /* Initialize the Resolution member */
    ADC_InitParam->Resolution = ADC_DATA_RES_12BIT;
}


/**
*\*\name    ADC_Enable.
*\*\fun     Configures the specified ADC enable or disable.
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void ADC_Enable(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        /* Set the AD_ON bit to wake up the ADC from power down mode */
        ADC->CTRL2 |= ADC_ON_EN_MASK;
    }
    else
    {
        /* Disable the selected ADC peripheral */
        ADC->CTRL2 &= (~ADC_ON_EN_MASK);
    }
}

/**
*\*\name    ADC_ConfigDiscModeChannelCount.
*\*\fun     Configures the discontinuous numbers for the selected ADC regular
*\*\        group channels.
*\*\param   Number : specifies the discontinuous mode regular channel
 *          count value. This number must be range form 1 to 8.
*\*\return  none
**/
void ADC_ConfigDiscModeChannelCount(uint8_t Number)
{
    __IO uint32_t tempreg;
    /* Get the old register value */
    tempreg = ADC->CTRL1;
    /* Clear the old discontinuous mode channel count */
    tempreg &= (~ADC_DISC_NUM_MASK);
    /* Set the discontinuous mode channel count */
    tempreg |= (((uint32_t)Number - 1U) << 13U);
    /* Store the new register value */
    ADC->CTRL1 = tempreg;
}

/**
*\*\name    ADC_ConfigRegularChannel.
*\*\fun     Configures for the selected ADC regular channel its corresponding
*\*\        rank in the sequencer and its sample time.
*\*\param   ADC_Channel :
*\*\          - ADC_CH_0 : ADC Channel0 selected
*\*\          - ADC_CH_1 : ADC Channel1 selected
*\*\          - ADC_CH_2 : ADC Channel2 selected
*\*\          - ADC_CH_3 : ADC Channel3 selected
*\*\          - ADC_CH_4 : ADC Channel4 selected
*\*\          - ADC_CH_5 : ADC Channel5 selected
*\*\          - ADC_CH_6 : ADC Channel6 selected
*\*\          - ADC_CH_7 : ADC Channel7 selected
*\*\          - ADC_CH_8 : ADC Channel8 selected
*\*\          - ADC_CH_9 : ADC Channel9 selected
*\*\          - ADC_CH_10 : ADC Channel10 selected
*\*\          - ADC_CH_11 : ADC Channel11 selected
*\*\          - ADC_CH_12 : ADC Channel12 selected
*\*\          - ADC_CH_13 : ADC Channel13 selected
*\*\          - ADC_CH_14 : ADC Channel14 selected
*\*\          - ADC_CH_15 : ADC Channel15 selected
*\*\          - ADC_CH_16 : ADC Channel16 selected
*\*\          - ADC_CH_17 : ADC Channel17 selected
*\*\          - ADC_CH_18 : ADC Channel18 selected
*\*\param   Rank : The rank in the regular group sequencer. This parameter must be between 1 to 16.
*\*\param   ADC_SampleTime : The sample time value to be set for the selected channel.
*\*\          When ADC_SAMPT3.SMPSEL = 0,select the sampling time below.
*\*\          - ADC_SAMP_TIME_CYCLES0_1_5     
*\*\          - ADC_SAMP_TIME_CYCLES0_7_5
*\*\          - ADC_SAMP_TIME_CYCLES0_13_5
*\*\          - ADC_SAMP_TIME_CYCLES0_28_5
*\*\          - ADC_SAMP_TIME_CYCLES0_41_5
*\*\          - ADC_SAMP_TIME_CYCLES0_55_5
*\*\          - ADC_SAMP_TIME_CYCLES0_71_5
*\*\          - ADC_SAMP_TIME_CYCLES0_239_5
*\*\          When ADC_SAMPT3.SMPSEL = 1,select the sampling time below.
*\*\          - ADC_SAMP_TIME_CYCLES1_1_5     
*\*\          - ADC_SAMP_TIME_CYCLES1_2_5
*\*\          - ADC_SAMP_TIME_CYCLES1_4_5
*\*\          - ADC_SAMP_TIME_CYCLES1_7_5
*\*\          - ADC_SAMP_TIME_CYCLES1_19_5
*\*\          - ADC_SAMP_TIME_CYCLES1_61_5
*\*\          - ADC_SAMP_TIME_CYCLES1_181_5
*\*\          - ADC_SAMP_TIME_CYCLES1_601_5
*\*\return  none
**/
void ADC_ConfigRegularChannel(uint8_t ADC_Channel, uint8_t Rank, uint8_t ADC_SampleTime)
{
    uint32_t tempreg1, tempreg2;
    if (ADC_Channel > ADC_CH_9)  /* if ADC_CH_10 ... ADC_CH_18 -> SAMPT1 */
    {
        /* Get the old register value */
        tempreg1 = ADC->SAMPT1;
        /* Calculate the mask to clear */
        tempreg2 = ADC_SAMP_TIME_CYCLES_MASK << (3U * (ADC_Channel - 10U));
        /* Clear the old channel sample time */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = ((uint32_t)ADC_SampleTime) << (3U * (ADC_Channel - 10U));
        /* Set the new channel sample time */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->SAMPT1 = tempreg1;
    }
    else /* if ADC_CH_0 ... ADC_CH_9 -> SAMPT2 */
    {
        /* Get the old register value */
        tempreg1 = ADC->SAMPT2;
        /* Calculate the mask to clear */
        tempreg2 = ADC_SAMP_TIME_CYCLES_MASK << (3U * (ADC_Channel));
        /* Clear the old channel sample time */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = ((uint32_t)ADC_SampleTime) << (3U * (ADC_Channel));
        /* Set the new channel sample time */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->SAMPT2 = tempreg1;
    }

    if (Rank < 7U)  /* For Rank 1 to 6 */
    {
        /* Get the old register value */
        tempreg1 = ADC->RSEQ3;
        /* Calculate the mask to clear */
        tempreg2 = ADC_RESQ_SEQ_MASK << (5U * (Rank - 1U));
        /* Clear the old SQx bits for the selected rank */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = (uint32_t)ADC_Channel << (5U * (Rank - 1U));
        /* Set the SQx bits for the selected rank */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->RSEQ3 = tempreg1;
    }
    else if (Rank < 13U)   /* For Rank 7 to 12 */
    {
        /* Get the old register value */
        tempreg1 = ADC->RSEQ2;
        /* Calculate the mask to clear */
        tempreg2 = ADC_RESQ_SEQ_MASK << (5U * (Rank - 7U));
        /* Clear the old SQx bits for the selected rank */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = (uint32_t)ADC_Channel << (5U * (Rank - 7U));
        /* Set the SQx bits for the selected rank */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->RSEQ2 = tempreg1;
    }
    else /* For Rank 13 to 16 */
    {
        /* Get the old register value */
        tempreg1 = ADC->RSEQ1;
        /* Calculate the mask to clear */
        tempreg2 = ADC_RESQ_SEQ_MASK << (5U * (Rank - 13U));
        /* Clear the old SQx bits for the selected rank */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = (uint32_t)ADC_Channel << (5U * (Rank - 13U));
        /* Set the SQx bits for the selected rank */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->RSEQ1 = tempreg1;
    }
}

/**
*\*\name    ADC_ConfigExternalTrigRegularConv.
*\*\fun     Configures the ADC external trigger source for regular channels conversion.
*\*\param   ADC_ExternalTrigRegularConv :
*\*\          - ADC_EXT_TRIG_REG_CONV_ATIM1_CC1
*\*\          - ADC_EXT_TRIG_REG_CONV_ATIM1_CC2
*\*\          - ADC_EXT_TRIG_REG_CONV_ATIM1_CC3
*\*\          - ADC_EXT_TRIG_REG_CONV_GTIM1_CC1
*\*\          - ADC_EXT_TRIG_REG_CONV_GTIM1_CC2
*\*\          - ADC_EXT_TRIG_REG_CONV_GTIM1_TRGO
*\*\          - ADC_EXT_TRIG_REG_CONV_EXTI_LINE0_15_OR_ATIM1_TRGO
*\*\          - ADC_EXT_TRIG_REG_CONV_SOFTWARE
*\*\return  none
**/
void ADC_ConfigExternalTrigRegularConv(uint32_t ADC_ExternalTrigRegularConv)
{
    uint32_t tempreg;

    /* Get the old register value */
    tempreg = ADC->CTRL2;
    /* Clear the old external event selection for regular group */
    tempreg &= (~ADC_EXT_TRIG_REG_CONV_MASK);
    /* Set the external event selection for regular group */
    tempreg |= ADC_ExternalTrigRegularConv;
    /* Store the new register value */
    ADC->CTRL2 = tempreg;
}

/**
*\*\name    ADC_EnableSoftwareStartConv.
*\*\fun     Enables or disables the selected ADC software start conversion .
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void ADC_EnableSoftwareStartConv(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        /* Enable the selected ADC conversion on external event and start the selected
           ADC conversion */
        ADC->CTRL2 |= ADC_REG_SWSTART_MASK;
    }
    else
    {
        /* Disable the selected ADC conversion on external event and stop the selected
           ADC conversion */
        ADC->CTRL2 &= (~ADC_REG_SWSTART_MASK);
    }
}


/**
*\*\name    ADC_GetSoftwareStartConvStatus.
*\*\fun     Gets the selected ADC Software start conversion Status.
*\*\return The new state of ADC software start conversion (SET or RESET).
**/
FlagStatus ADC_GetSoftwareStartConvStatus(void)
{
    FlagStatus bitstatus;

    if ((ADC->CTRL2 & ADC_REG_SWSTART_MASK) != (uint32_t)RESET)
    {
        /* SOFT_START bit is set */
        bitstatus = SET;
    }
    else
    {
        /* SOFT_START bit is reset */
        bitstatus = RESET;
    }
    return bitstatus;
}

}


/**
*\*\name    ADC_EnableTempSensor.
*\*\fun     Enables or disables the temperature sensor channel.
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void ADC_EnableTempSensor(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        ADC->CTRL3 |= ADC_CTRL3_BUFF_EN;
        /* Enable the temperature sensor channel*/
        ADC->CTRL2 |= (ADC_CTRL2_VREFINT_TS_SEL | ADC_TS_EN_MASK);
    }
    else
    {
        /* Disable the temperature sensor channel*/
        ADC->CTRL2 &= (~ADC_TS_EN_MASK);
        ADC->CTRL3 &= ~ADC_CTRL3_BUFF_EN;
    }
}

/**
*\*\name    ADC_EnableVrefint.
*\*\fun     Enables or disables the Vrefint channel.
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void ADC_EnableVrefint(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        ADC->CTRL3 |= ADC_CTRL3_BUFF_EN;
        /* Select vrefint channel and Enable it*/
        ADC->CTRL2 &= ~ADC_CTRL2_VREFINT_TS_SEL;
    }
    else
    {
        /* Disable the vrefint channel channel*/
        ADC->CTRL3 &= ~ADC_CTRL3_BUFF_EN;
    }
}

*\*\name    ADC_ConfigInjectedChannel.

*\*\          - ADC_CH_4 : ADC Channel4 selected
*\*\          - ADC_CH_5 : ADC Channel5 selected
*\*\          - ADC_CH_6 : ADC Channel6 selected
*\*\          - ADC_CH_7 : ADC Channel7 selected
*\*\          - ADC_CH_8 : ADC Channel8 selected
*\*\          - ADC_CH_9 : ADC Channel9 selected
*\*\          - ADC_CH_10 : ADC Channel10 selected
*\*\          - ADC_CH_11 : ADC Channel11 selected
*\*\          - ADC_CH_12 : ADC Channel12 selected

*\*\          When ADC_SAMPT3.SMPSEL = 0,select the sampling time below.
*\*\          - ADC_SAMP_TIME_CYCLES0_1_5     
*\*\          - ADC_SAMP_TIME_CYCLES0_7_5
*\*\          - ADC_SAMP_TIME_CYCLES0_13_5
*\*\          - ADC_SAMP_TIME_CYCLES0_28_5
*\*\          - ADC_SAMP_TIME_CYCLES0_41_5
*\*\          - ADC_SAMP_TIME_CYCLES0_55_5
*\*\          - ADC_SAMP_TIME_CYCLES0_71_5
*\*\          - ADC_SAMP_TIME_CYCLES0_239_5
*\*\          When ADC_SAMPT3.SMPSEL = 1,select the sampling time below.
*\*\          - ADC_SAMP_TIME_CYCLES1_1_5     
*\*\          - ADC_SAMP_TIME_CYCLES1_2_5
*\*\          - ADC_SAMP_TIME_CYCLES1_4_5
*\*\          - ADC_SAMP_TIME_CYCLES1_7_5
*\*\          - ADC_SAMP_TIME_CYCLES1_19_5
*\*\          - ADC_SAMP_TIME_CYCLES1_61_5
*\*\          - ADC_SAMP_TIME_CYCLES1_181_5
*\*\          - ADC_SAMP_TIME_CYCLES1_601_5
*\*\return  none
**/
void ADC_ConfigInjectedChannel(uint8_t ADC_Channel, uint8_t Rank, uint8_t ADC_SampleTime)
{
    uint32_t tempreg1, tempreg2, tempreg3;

    if (ADC_Channel > ADC_CH_9)  /* ADC_CH_10 ... ADC_CH_18 -> SAMPT1 */
    {
        /* Get the old register value */
        tempreg1 = ADC->SAMPT1;
        /* Calculate the mask to clear */
        tempreg2 = ADC_SAMP_TIME_CYCLES_MASK << (3U * (ADC_Channel - 10U));
        /* Clear the old channel sample time */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = ((uint32_t)ADC_SampleTime) << (3U * (ADC_Channel - 10U));
        /* Set the new channel sample time */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->SAMPT1 = tempreg1;
    }
    else  /* ADC_CH_0 ... ADC_CH_9 -> SAMPT2 */
    {
        /* Get the old register value */
        tempreg1 = ADC->SAMPT2;
        /* Calculate the mask to clear */
        tempreg2 = ADC_SAMP_TIME_CYCLES_MASK << (3U * ADC_Channel);
        /* Clear the old channel sample time */
        tempreg1 &= ~tempreg2;
        /* Calculate the mask to set */
        tempreg2 = ((uint32_t)ADC_SampleTime) << (3U * ADC_Channel);
        /* Set the new channel sample time */
        tempreg1 |= tempreg2;
        /* Store the new register value */
        ADC->SAMPT2 = tempreg1;
    }
    /* Rank configuration */
    /* Get the old register value */
    tempreg1 = ADC->JSEQ;
    /* Get JLEN value: Number = JLEN+1 */
    tempreg3 = (tempreg1 & ADC_JESQ_LEN_MASK) >> 20U;
    /* Calculate the mask to clear: ((Rank-1)+(4-JLEN-1)) */
    tempreg2 = ADC_JESQ_SEQ_MASK << (5U * (((uint32_t)Rank + 3U) - (tempreg3 + 1U)));
    /* Clear the old JSEQx bits for the selected rank */
    tempreg1 &= ~tempreg2;
    /* Calculate the mask to set: ((Rank-1)+(4-JLEN-1)) */
    tempreg2 = (uint32_t)ADC_Channel << (5U * (((uint32_t)Rank + 3U) - (tempreg3 + 1U)));
    /* Set the JSEQx bits for the selected rank */
    tempreg1 |= tempreg2;
    /* Store the new register value */
    ADC->JSEQ = tempreg1;
}

/**
*\*\name    ADC_SetOffsetConfig.
*\*\fun     Set the specified channel conversion offset configuration.
*\*\param   ADC_Offset :
*\*\          - ADC_REGESTER_OFFSET_1
*\*\          - ADC_REGESTER_OFFSET_2
*\*\          - ADC_REGESTER_OFFSET_3
*\*\          - ADC_REGESTER_OFFSET_4
*\*\param   ADC_OffsetStruct :
*\*\          - OffsetData : Set the selected channel data offset .This parameter must be range from 0 to 0xfff.
*\*\          - OffsetChannel : Set the selected channel .This parameter must be range from 0 to 0x12. means CH0 - CH18.
*\*\          - OffsetDirPositiveEn : Enable or disable the selected channel offset positive direction.
*\*\             - ENABLE
*\*\             - DISABLE
*\*\          - OffsetSatenEn : Enable or disable the selected channel offset saturation .
*\*\             - ENABLE
*\*\             - DISABLE
*\*\          - OffsetEn : Enable or disable the selected channel offset.
*\*\             - ENABLE
*\*\             - DISABLE
*\*\return  none
**/
void ADC_SetOffsetConfig(uint8_t ADC_Offset, const ADC_OffsetType* ADC_OffsetStruct)
{
    uint32_t tempreg, temp;

    tempreg = (uint32_t)(uintptr_t)ADC;
    tempreg += ADC_Offset;
    /* Get register value */
    temp = *((__IO uint32_t*)(uintptr_t)tempreg);
    temp &= (~(ADC_OFFSET_DATA_MASK | ADC_OFFSET_EN_MASK | ADC_OFFSET_CH_MASK | ADC_OFFSET_SATEN_EN_MASK| ADC_OFFSET_DIR_MASK ));
    /* Set the selected channel offset direction and channel */
    temp |= ((uint32_t)ADC_OffsetStruct->OffsetData | (((uint32_t)ADC_OffsetStruct->OffsetChannel) << (uint32_t)26U));

    /* Set the selected channel offset direction, saturation, enable or disable*/
    temp |= (BIT_JUDGE_AND_HANDLER(ADC_OffsetStruct->OffsetDirPositiveEn,ADC_OFFSET_DIR_MASK) | \
             BIT_JUDGE_AND_HANDLER(ADC_OffsetStruct->OffsetSatenEn,ADC_OFFSET_SATEN_EN_MASK) | \
             BIT_JUDGE_AND_HANDLER(ADC_OffsetStruct->OffsetEn,ADC_OFFSET_EN_MASK));

    *((__IO uint32_t*)(uintptr_t)tempreg) = temp;
}

/**
*\*\name    ADC_ConfigInt.
*\*\fun     Enables or disables the specified ADC interrupts.
*\*\param   ADC_IT :
*\*\          - ADC_INT_ENDC
*\*\          - ADC_INT_ENDCA
*\*\          - ADC_INT_JENDC
*\*\          - ADC_INT_JENDCA
*\*\          - ADC_INT_AWDG
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return    none.
**/
void ADC_ConfigInt(uint16_t ADC_IT, FunctionalState Cmd)
{
    uint32_t itmask;
    /* Get the ADC IT */
    itmask = (uint32_t)((ADC_IT & 0xFF00U) >> 4U);

    if (Cmd != DISABLE)
    {
        /* Enable the selected ADC interrupts */
        if ((itmask & ADC_INT_CTRL1_MASK) != 0U)
        {
            ADC->CTRL1 |= (itmask & ADC_INT_CTRL1_MASK);
        }
        if ((itmask & ADC_INT_CTRL3_MASK) != 0U)
        {
            ADC->CTRL3 |= (itmask & ADC_INT_CTRL3_MASK);
        }
    }
    else
    {
        /* Disable the selected ADC interrupts */
        if ((itmask & ADC_INT_CTRL1_MASK) != 0U)
        {
            ADC->CTRL1 &= (~(itmask & ADC_INT_CTRL1_MASK));
        }
        if ((itmask & ADC_INT_CTRL3_MASK) != 0U)
        {
            ADC->CTRL3 &= (~(itmask & (ADC_INT_CTRL3_MASK)));
        }
    }
}


/**
*\*\name    ADC_GetIntStatus.
*\*\fun     Checks whether the specified ADC interrupt has occurred or not.
*\*\param   ADC_IT : specifies the ADC interrupt source to check.
*\*\          - ADC_INT_ENDC
*\*\          - ADC_INT_ENDCA
*\*\          - ADC_INT_JENDC
*\*\          - ADC_INT_JENDCA
*\*\          - ADC_INT_AWDG
*\*\return    The new state of ADC_IT (SET or RESET).
**/
INTStatus ADC_GetIntStatus(uint16_t ADC_IT)
{
    INTStatus bitstatus;
    uint32_t itmask, enablestatus;

    itmask = (uint32_t)((ADC_IT & 0xFF00U) >> 4U);
    /* Get the ADC_IT enable bit status from CTRL1 or CTRL3 */
    if ((itmask & ADC_INT_CTRL1_MASK) != 0U)
    {
        enablestatus = (ADC->CTRL1 & itmask);
    }
    else
    {
        enablestatus = (ADC->CTRL3 & itmask);
    }
    itmask = (uint32_t)(ADC_IT & 0x00FFU);
    /* Check the status of the specified ADC interrupt */
    if (((ADC->STS & itmask) != 0U) && (enablestatus != 0U))
    {
        /* ADC_IT is set */
        bitstatus = SET;
    }
    else
    {
        /* ADC_IT is reset */
        bitstatus = RESET;
    }
    return bitstatus;
}

/**
*\*\name    ADC_GetCalibrationStatus.
*\*\fun     Gets the selected ADC calibration status.
*\*\param   cali_mode : specifies differential or signal mode of calibration.
*\*\          - ADC_CALIBRATION_SINGLE_MODE
*\*\          - ADC_CALIBRATION_DIFF_MODE
*\*\return  FlagStatus:
*\*\          - RESET : ADC calibration is finished;
*\*\          - SET   : ADC calibration is not finished;
**/
FlagStatus ADC_GetCalibrationStatus(ADC_CALI_MOD cali_mode)
{
    FlagStatus bitstatus;

    /* Check the status of CAL bit */
    if ((ADC->CTRL2 & ADC_CALI_EN_MASK) != (uint32_t)RESET)
    {
        /* CAL bit is set: calibration on going */
        bitstatus = SET;
    }
    else
    {
        /* CAL bit is reset: end of calibration */
        bitstatus = RESET;
    }
    if (cali_mode == ADC_CALIBRATION_DIFF_MODE)
    {
        if ((ADC->CALFACT&ADC_CALFACT_CALFACTD_MASK) != 0U)
        {
            bitstatus = RESET;
        }
        else
        {
            /* no process*/
        }
    }
    else
    {
        if ((ADC->CALFACT&ADC_CALFACT_CALFACTS_MASK) != 0U)
        {
            bitstatus = RESET;
        }
        else
        {
            /* no process*/
        }
    }
    /* Return the CAL bit status */
    return bitstatus;
}

*\*\name    ADC_SetCalibrationFactor.
*\*\fun     Set ADC calibration factor in the single-ended or differential mode.
*\*\param   SingleDiff : 
*\*\          - ADC_SINGLE_ENDED
*\*\          - ADC_DIFFERENTIAL_ENDED
*\*\param   CalibrationFactor : The parameter between Min_Data=0x00 and Max_Data=0x7F
*\*\return    none.
**/
void ADC_SetCalibrationFactor(uint32_t SingleDiff, uint32_t CalibrationFactor)
{
    if (SingleDiff == ADC_DIFFERENTIAL_ENDED)
    {
        ADC->CALFACT &= (~(ADC_CALFACT_CALFACTD_MASK));
        ADC->CALFACT |= (CalibrationFactor << 16U);
    }
    else
    {
        ADC->CALFACT &= (~(ADC_CALFACT_CALFACTS_MASK));
        ADC->CALFACT |= CalibrationFactor;
    }
}


/**
*\*\name    ADC_GetCalibrationFactor.
*\*\fun     Reads the calibration factors from the CALFACT register.
*\*\param   CalFactS : pointer to store the single-ended calibration factor (7-bit).
*\*\param   CalFactD : pointer to store the differential calibration factor (7-bit).
*\*\return  none
**/
void ADC_GetCalibrationFactor(uint8_t* CalFactS, uint8_t* CalFactD)
{
    uint32_t tempreg;

    tempreg = ADC->CALFACT;
    *CalFactS = (uint8_t)(tempreg & ADC_CALFACT_CALFACTS_MASK);
    *CalFactD = (uint8_t)((tempreg & ADC_CALFACT_CALFACTD_MASK) >> 16U);
}


/**
*\*\name    ADC_SetChannelSingleDiff.
*\*\fun     Set mode single-ended or differential input of the selected ADC channel.
*\*\param   Channel : 
*\*\          - ADC_CH_1
*\*\          - ADC_CH_2
*\*\          - ADC_CH_3
*\*\          - ADC_CH_4 
*\*\          - ADC_CH_5
*\*\          - ADC_CH_6
*\*\          - ADC_CH_7
*\*\          - ADC_CH_8 
*\*\          - ADC_CH_9 
*\*\          - ADC_CH_10
*\*\          - ADC_CH_11
*\*\          - ADC_CH_12
*\*\          - ADC_CH_13
*\*\          - ADC_CH_14
*\*\          - ADC_CH_15
*\*\          - ADC_CH_16
*\*\          - ADC_CH_17
*\*\          - ADC_CH_18
*\*\param   SingleDiff : 
*\*\          - ADC_SINGLE_ENDED
*\*\          - ADC_DIFFERENTIAL_ENDED
*\*\return    none.
*\*\note  Some channels are internally fixed to single-ended inputs: please refer to ADC chapter of user manual.
**/
void ADC_SetChannelSingleDiff(uint32_t Channel, uint32_t SingleDiff)
{
    if (SingleDiff == ADC_DIFFERENTIAL_ENDED)
    {
        ADC->DIFSEL |= ((uint32_t)0x01U << Channel);
    }
    else
    {
        ADC->DIFSEL &= (~((uint32_t)0x01U << Channel));
    }
}

/**
 *\*\name   ADC_ConfigADCPowerSupport.
 *\*\fun    Configurate ADC power support ,select the external MR or internal LDO.
 *\*\param  ADC_PowerSupportSel : 
 *\*\          - ADC_EXTERN_POWER_SUPPORT , select MR power support 
 *\*\          - ADC_INTERN_POWER_SUPPORT , select LDO power support 
 *\*\return none
 */
void ADC_ConfigADCPowerSupport(uint32_t ADC_PowerSupportSel)
{
    uint32_t tempreg;

    /* Get the old register value */
    tempreg = ADC->PCTRL;
    /* clear ADC_PCTRL register EXPEN RANGESEL VREFLDOEN bit */
    tempreg &= (~(ADC_PCTRL_EXPEN_MASK | ADC_PCTRL_RANGESEL_MASK | ADC_PCTRL_VREFLDOEN_MASK));
    /* Set EXPEN RANGESEL VREFLDOEN select bits */
    tempreg |= ADC_PowerSupportSel;
     /* Store the new register value */
    ADC->PCTRL = tempreg;
}


/**
*\*\name    ADC_EnableDeepSleepMode.
*\*\fun     Enables or disables the specified ADC entering deep sleep mode.
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void ADC_EnableDeepSleepMode(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        /* Enable the selected ADC entering deep sleep mode */
        ADC->CTRL3 |= ADC_CTRL3_DPWMOD;
    }
    else
    {
        /* Disable the selected ADC entering deep sleep mode */
        ADC->CTRL3 &= (~ADC_CTRL3_DPWMOD);
    }
}

/**
*\*\name    ADC_ConfigSampleTimeSelect.
*\*\fun     Configure the channel sampling time select bit.
*\*\param   Cmd :
*\*\          - ENABLE  (set SMPSEL=1, use alternative sample time table)
*\*\          - DISABLE (set SMPSEL=0, use default sample time table)
*\*\return  none
**/
void ADC_ConfigSampleTimeSelect(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        ADC->SAMPT3 |= ADC_SAMPT3_SMPSEL;
    }
    else
    {
        ADC->SAMPT3 &= (~ADC_SAMPT3_SMPSEL);
    }
}

/**
*\*\name    ADC_ConfigClk.
*\*\fun     Configures the ADC clock.
*\*\param   ADC_ClkMode :
*\*\         - ADC_CKMOD_HSI_DIV2
*\*\         - ADC_CKMOD_HSE
*\*\param   ADC_CLKPre : specifies the ADC_CLK prescaler.
*\*\         - RCC_ADCACLK_DIV1        ADCACLK Clock Divided By 1 
*\*\         - RCC_ADCACLK_DIV2        ADCACLK Clock Divided By 2 
*\*\         - RCC_ADCACLK_DIV4        ADCACLK Clock Divided By 4 
*\*\         - RCC_ADCACLK_DIV6        ADCACLK Clock Divided By 6 
*\*\         - RCC_ADCACLK_DIV8        ADCACLK Clock Divided By 8 
*\*\         - RCC_ADCACLK_DIV10       ADCACLK Clock Divided By 10
*\*\         - RCC_ADCACLK_DIV12       ADCACLK Clock Divided By 12
*\*\         - RCC_ADCACLK_DIV16       ADCACLK Clock Divided By 16
*\*\         - RCC_ADCACLK_DIV32       ADCACLK Clock Divided By 32
*\*\         - RCC_ADCACLK_DIV64        ADCACLK Clock Divided By 64  
*\*\param   ADC_1MCLKPre : specifies the ADC_1MCLK prescaler.
*\*\         - RCC_ADC1MCLK_DIV1        ADC1MCLK Clock Divided By 1 
*\*\         - RCC_ADC1MCLK_DIV2        ADC1MCLK Clock Divided By 2 
*\*\         - RCC_ADC1MCLK_DIV4        ADC1MCLK Clock Divided By 4 
*\*\         - RCC_ADC1MCLK_DIV6        ADC1MCLK Clock Divided By 6 
*\*\         - RCC_ADC1MCLK_DIV8        ADC1MCLK Clock Divided By 8 
*\*\         - RCC_ADC1MCLK_DIV10       ADC1MCLK Clock Divided By 10
*\*\         - RCC_ADC1MCLK_DIV12       ADC1MCLK Clock Divided By 12
*\*\         - RCC_ADC1MCLK_DIV16       ADC1MCLK Clock Divided By 16
*\*\         - RCC_ADC1MCLK_DIV32       ADC1MCLK Clock Divided By 32
*\*\         - RCC_ADC1MCLK_DIV64       ADC1MCLK Clock Divided By 64
*\*\return    none.
**/
void ADC_ConfigClk(ADC_CLKSRC ADC_ClkMode, uint32_t ADC_CLKPre, uint32_t ADC_1MCLKPre)
{
    if (ADC_ClkMode == ADC_CKMOD_HSI_DIV2)
    {
        RCC_ConfigAdcClk( RCC_ADCCLK_SRC_HSI_DIV2);
    }
    else
    {
        RCC_ConfigAdcClk( RCC_ADCCLK_SRC_HSE);
    }
    RCC_ConfigAdcAclkPre(ADC_CLKPre);
    RCC_ConfigAdc1MclkPre(ADC_1MCLKPre);   
}
