#include "misc.h"

/** MISC Driving Functions Declaration **/

/**
*\*\name    NVIC_Init
*\*\fun     Initializes the NVIC peripheral according to the specified parameters in the NVIC_structure_initializes.
*\*\param   NVIC_InitStruct (The input parameters must be the following values) :
*\*\              - NVIC_IRQChannel :
*\*\                   -  IRQn_Type    N32WB033 Interrupt Number Definition
*\*\              - NVIC_IRQChannelPriority :
*\*\                   -  NVIC_PRIORITY_0
*\*\                   -  NVIC_PRIORITY_1
*\*\                   -  NVIC_PRIORITY_2
*\*\                   -  NVIC_PRIORITY_3
*\*\              - NVIC_IRQChannelCmd :
*\*\                   -  ENABLE
*\*\                   -  DISABLE
*\*\return  none
**/
void NVIC_Init(const NVIC_InitType* NVIC_InitStruct)
{
    /* Set Interrupt Priority */
    NVIC_SetPriority(NVIC_InitStruct->NVIC_IRQChannel, NVIC_InitStruct->NVIC_IRQChannelPriority);

    if (NVIC_InitStruct->NVIC_IRQChannelCmd != DISABLE)
    {
        /* Enable the Selected IRQ Channels */
        NVIC_EnableIRQ(NVIC_InitStruct->NVIC_IRQChannel);
    }
    else
    {
        /* Disable the Selected IRQ Channels */
        NVIC_DisableIRQ(NVIC_InitStruct->NVIC_IRQChannel);
    }
}


/**
*\*\name    NVIC_SetVectorTable
*\*\fun     Sets the vector table location and Offset.
*\*\param   NVIC_vecter_table specifies if the vector table is in RAM or FLASH memory.
*\*\        This parameter can be one of the following values:
*\*\            - NVIC_VectTab_RAM
*\*\            - NVIC_VectTab_FLASH
*\*\param   offset Vector Table base offset field. This value must be a multiple of 0x200.
*\*\return  none
**/
void NVIC_SetVectorTable(uint32_t NVIC_VectTab, uint32_t Offset)
{
    SCB->VTOR = NVIC_VectTab | (Offset & (uint32_t)0x1FFFFF80U);
}


/**
*\*\name  SysTick_CLKSourceConfig
*\*\fun   Configures the SysTick clock source.
*\*\param SysTick_CLKSource (The input parameters must be the following values) :
*\*\       - SysTick_CLKSource_HCLK_Div8    External clock selected as SysTick clock source.
*\*\       - SysTick_CLKSource_HCLK         AHB clock selected as SysTick clock source.
*\*\return  none
**/
void SysTick_CLKSourceConfig(uint32_t SysTick_CLKSource)
{
    if (SysTick_CLKSource == SysTick_CLKSource_HCLK)
    {
        SysTick->CTRL |= SysTick_CLKSource_HCLK;
    }
    else
    {
        SysTick->CTRL &= SysTick_CLKSource_HCLK_Div8;
    }
}
