/**
 *\*\name   flash_erase.
 *\*\fun    Erase a flash sector at the specified address.
 *\*\param  flash_address :
 *\*\           absolute memory address of the sector to erase (must be sector-aligned 4KB).
 *\*\return operation status (ReturnMsg).
 */
ReturnMsg flash_erase(uint32_t flash_address)
{
    return erase(flash_address - 0x01000000U);
}

/**
 *\*\name   flash_program.
 *\*\fun    Program data to flash memory (page program).
 *\*\param  flash_address :
 *\*\           absolute memory address of the destination (should be page-aligned).
 *\*\param  source_address :
 *\*\           pointer to the source data buffer.
 *\*\param  byte_length :
 *\*\           number of bytes to program (must not exceed page size 256 Bytes).
 *\*\return operation status (ReturnMsg).
 */
ReturnMsg flash_program(uint32_t flash_address, const uint8_t *source_address, uint32_t byte_length)
{
    return program(flash_address - 0x01000000U, source_address, byte_length);
}

/**
 *\*\name   flash_read.
 *\*\fun    Read data from flash memory.
 *\*\param  flash_address :
 *\*\           absolute memory address of the data to read.
 *\*\param  target_address :
 *\*\           pointer to the destination buffer.
 *\*\param  byte_length :
 *\*\           number of bytes to read.
 *\*\return operation status (ReturnMsg).
 */
ReturnMsg flash_read(uint32_t flash_address, uint8_t *target_address, uint32_t byte_length)
{
    return read(flash_address - 0x01000000U, target_address, byte_length);
}

/**
 *\*\name   flash_otp_read.
 *\*\fun    Read data from OTP security registers.
 *\*\param  security_reg_num :
 *\*\           OTP security register index.
 *\*\param  buffer :
 *\*\           pointer to the destination buffer.
 *\*\param  buffer_size :
 *\*\           number of bytes to read.
 *\*\return operation status (ReturnMsg).
 */
ReturnMsg flash_otp_read(uint8_t security_reg_num, uint8_t* buffer, uint32_t buffer_size)
{
    return otp_read(security_reg_num, buffer, buffer_size);
}
