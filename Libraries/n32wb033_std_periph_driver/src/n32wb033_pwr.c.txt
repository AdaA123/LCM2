/** PWR Driving Functions Declaration **/


/**
*\*\name    PWR_DeInit.
*\*\fun     Deinitializes the PWR peripheral registers to their default reset values.
*\*\param   none
*\*\return  none
**/
void PWR_DeInit(void)
{
    RCC_EnableAPB1PeriphReset(RCC_APB1_PERIPH_PWR);
}

/**
*\*\name    PWR_EnterIDLEMode.
*\*\fun     Enters IDLE mode.
*\*\param   IDLEONEXIT :
*\*\          - DISABLE    SLEEP mode with SLEEPONEXIT disable
*\*\          - ENABLE     SLEEP mode with SLEEPONEXIT enable
*\*\param   PWR_IDLEEntry :
*\*\          - PWR_IDLEENTRY_WFI    enter IDLE mode with WFI instruction
*\*\          - PWR_IDLEENTRY_WFE    enter IDLE mode with WFE instruction
*\*\return  none
**/
void PWR_EnterIDLEMode(uint8_t IDLEONEXIT, uint8_t PWR_IDLEEntry)
{

    /* CLEAR SLEEPDEEP bit of Cortex System Control Register */
    SCB->SCR &= (uint32_t)~((uint32_t)SCB_SCR_SLEEPDEEP);

      /* Select SLEEPONEXIT mode entry --------------------------------------------------*/
    if (IDLEONEXIT == (uint8_t)ENABLE)
    {
        /* the MCU enters Sleep mode as soon as it exits the lowest priority ISR */
        SCB->SCR |= SCB_SCR_SLEEPONEXIT;
    }
    else if (IDLEONEXIT == (uint8_t)DISABLE)
    {
        /* Sleep-now */
        SCB->SCR &= (uint32_t)~((uint32_t)SCB_SCR_SLEEPONEXIT);
    }
    else
    {
    	/* No Process */
    }

    /* Select SLEEP mode entry --------------------------------------------------*/
    if (PWR_IDLEEntry == PWR_IDLEENTRY_WFI)
    {
        /* Request Wait For Interrupt */
        __WFI();
    }
    else
    {
        /* Request Wait For Event */
        __SEV();
        __WFE();
        __WFE();
    }
}

/**
*\*\name    PWR_EnterSLEEPMode.
*\*\fun     Enters SLEEP mode.
*\*\param   PWR_SleepEntry :
*\*\          - PWR_SLEEPENTRY_WFI    enter SLEEP mode with WFI instruction
*\*\          - PWR_SLEEPENTRY_WFE    enter SLEEP mode with WFE instruction
*\*\return  none
**/
 void PWR_EnterSLEEPMode(uint8_t PWR_SleepEntry)
{
    uint32_t tmpreg = 0U;
    /* Set BLE modem sleep */
//    *(volatile uint32_t*)(uintptr_t)0x40011804U |= 0x40U; /* enable this config for without HSE */
    *(volatile uint32_t*)(uintptr_t)(BASEBAND_REG_BASE + 0x30U) = 0x07U;
    while ((PWR->CTRL1 & PWR_CTRL1_BLE_OSC_EN) == PWR_CTRL1_BLE_OSC_EN)
    {
        /* wait for BLE oscillator to be disabled */
    }

    /* Select the regulator state in SLEEP mode ---------------------------------*/
    tmpreg = PWR->CTRL1;
    /* Clear PWR_MODE and PWR_MODE_EN bits */
    tmpreg &= ~(PWR_CTRL1_PWR_MODE|PWR_CTRL1_PWR_MODE_EN);
    /* Set PWR_MODE and PWR_MODE_EN bit */
    tmpreg |= (PWR_CTRL1_PWR_MODE_SLEEP | PWR_CTRL1_PWR_MODE_EN);
    /* Store the new value */
    PWR->CTRL1 = tmpreg;
    /* Set SLEEPDEEP bit of Cortex System Control Register */
    SCB->SCR |= SCB_SCR_SLEEPDEEP;

    /* Select SLEEP mode entry --------------------------------------------------*/
    if (PWR_SleepEntry == PWR_SLEEPENTRY_WFI)
    {
        /* Request Wait For Interrupt */
        __WFI();
    }
    else
    {
        /* Request Wait For Event */
        __SEV();
        __WFE();
        __WFE();
    }

    /* Reset SLEEPDEEP bit of Cortex System Control Register */
    SCB->SCR &= (uint32_t)~((uint32_t)SCB_SCR_SLEEPDEEP);
}

/**
*\*\name    PWR_EnableSleepIwdgFrozen
*\*\fun     Freeze the IWDG counter in Sleep mode.
*\*\param   Cmd (The input parameters must be the following values):
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void PWR_EnableSleepIwdgFrozen(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        PWR->IWDG_CTRL &= (~PWR_IWDG_CTRL_SLEEP_FRZ);
    }
    else
    {
        PWR->IWDG_CTRL |= PWR_IWDG_CTRL_SLEEP_FRZ;
    }
}

/**
*\*\name    PWR_EnableIdleIwdgFrozen
*\*\fun     Freeze the IWDG counter in Idle mode.
*\*\param   Cmd (The input parameters must be the following values):
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void PWR_EnableIdleIwdgFrozen(FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        PWR->IWDG_CTRL &= (~PWR_IWDG_CTRL_IDLE_FRZ);
    }
    else
    {
        PWR->IWDG_CTRL |= PWR_IWDG_CTRL_IDLE_FRZ;
    }
}
