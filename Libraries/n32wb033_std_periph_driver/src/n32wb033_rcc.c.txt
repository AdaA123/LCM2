/** RCC Driving Functions Declaration **/
/**
*\*\name    RCC_DeInit.
*\*\fun     Resets the RCC clock configuration to the default reset state.
*\*\param   none
*\*\return  none
**/
void RCC_DeInit(void)
{
    /* Set HSIEN bit */
    RCC->CTRL |= (uint32_t)0x00000001U;

    /* Reset SCLKSW, AHBPRE, APB1PRES, APB2PRES, MCO, MCOPRE bits */
    RCC->CFG &= (uint32_t)0x81FFC0CEU;

    /* Reset HSEEN bits */
    RCC->CTRL &= (uint32_t)0xFFFEFFFFU;

    /* Disable all interrupts and clear pending bits  */
    RCC->CLKINT = 0x001F0000U;
}

/**
*\*\name    RCC_ConfigHse.
*\*\fun     Configures the External High Speed oscillator (HSE).
*\*\param   RCC_HSE :
*\*\          - RCC_HSE_DISABLE    HSE oscillator OFF
*\*\          - RCC_HSE_ENABLE     HSE oscillator ON
*\*\return  none
*\*\note    HSE can not be stopped if it is used directly as system clock.
**/
void RCC_ConfigHse(uint32_t RCC_HSE)
{
    /* Reset HSEON and HSEIOSEL bits before configuring the HSE ------------------*/
    /* Reset HSEON bit */
    RCC->CTRL &= ~RCC_CTRL_HSEEN;
    /* Configure HSE (RCC_HSE_DISABLE is already covered by the code section above) */
    switch (RCC_HSE)
    {
        case RCC_HSE_ENABLE:
            /* Set HSEEN bit */
            RCC->CTRL |= RCC_CTRL_HSEEN;
            break;
        default:
        	/* No Process */
            break;
    }
}

/**
*\*\name    RCC_WaitHseStable.
*\*\fun     Waits for HSE start-up.
*\*\param   none
*\*\return  ErrorStatus:
 *\*\         - SUCCESS    HSE oscillator is stable and ready to use
 *\*\         - ERROR      HSE oscillator not yet ready
**/
ErrorStatus RCC_WaitHseStable(void)
{
    __IO uint32_t StartUpCounter = 0U;
    ErrorStatus status           = ERROR;
    FlagStatus HSEStatus         = RESET;

    /* Wait till HSE is ready and if Time out is reached exit */
    do
    {
        HSEStatus = RCC_GetFlagStatus(RCC_CTRL_FLAG_HSERDF);
        StartUpCounter += 1U;
    } while ((StartUpCounter != HSE_STARTUP_TIMEOUT) && (HSEStatus == RESET));

    if (RCC_GetFlagStatus(RCC_CTRL_FLAG_HSERDF) != RESET)
    {
        status = SUCCESS;
    }
    else
    {
        status = ERROR;
    }
    return (status);
}

/**
*\*\name    RCC_ConfigHsi.
*\*\fun     Configures the Internal High Speed oscillator (HSI).
*\*\param   RCC_HSI :
*\*\          - RCC_HSI_DISABLE    HSI oscillator OFF
*\*\          - RCC_HSI_ENABLE     HSI oscillator ON
*\*\return  none
*\*\note    HSI can not be stopped if it is used directly as system clock.
**/
void RCC_ConfigHsi(uint32_t RCC_HSI)
{
    /* Reset HSIEN bit */
    RCC->CTRL &= ~RCC_CTRL_HSIEN;
    /* Configure HSI */
    switch (RCC_HSI)
    {
        case RCC_HSI_ENABLE:
            /* Set HSIEN bit */
            RCC->CTRL |= RCC_CTRL_HSIEN;
            break;

        default:

            break;
    }
}

/**
*\*\name    RCC_EnableHsi.
*\*\fun     Enables or disables the Internal High Speed oscillator (HSI).
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
*\*\note    HSI can not be stopped if it is used directly as system clock.
**/
void RCC_EnableHsi(FunctionalState Cmd)
{
   if(Cmd == ENABLE)
   {
       /* Set HSIEN bit */
       RCC->CTRL |= RCC_CTRL_HSIEN;
   }
   else
   {
       /* Reset HSIEN bit */
       RCC->CTRL &= ~RCC_CTRL_HSIEN;
   }
}


/**
*\*\name    RCC_ConfigSysclk.
*\*\fun     Configures the system clock (SYSCLK).
*\*\param   RCC_SYSCLKSource :
*\*\          - RCC_SYSCLK_SRC_HSI    HSI selected as system clock
*\*\          - RCC_SYSCLK_SRC_HSE    HSE selected as system clock
*\*\return  none
**/
void RCC_ConfigSysclk(uint32_t RCC_SYSCLKSource)
{
    uint32_t tmpregister = 0U;
    
    tmpregister = RCC->CFG;
    /* Clear SCLKSW bits */
    tmpregister &= ~RCC_CFG_SCLKSW;
    /* Set SCLKSW bits according to RCC_SYSCLKSource value */
    tmpregister |= RCC_SYSCLKSource;
    /* Store the new value */
    RCC->CFG = tmpregister;
}

/**
*\*\name    RCC_GetSysclkSrc.
*\*\fun     Returns the clock source used as system clock.
*\*\param   none
*\*\return  The clock source used as system clock:
 *\*\         - 0x00000000: HSI used as system clock
 *\*\         - 0x00000004: HSE used as system clock
**/
uint32_t RCC_GetSysclkSrc(void)
{
    return ((uint32_t)(RCC->CFG & RCC_CFG_SCLKSTS));
}

/**
*\*\name    RCC_ConfigInt.
*\*\fun     Enables or disables the specified RCC interrupts.
*\*\param   RccInt :
*\*\          - RCC_INT_LSIRDIF    LSI ready interrupt
*\*\          - RCC_INT_LSERDIF    LSE ready interrupt
*\*\          - RCC_INT_HSIRDIF    HSI ready interrupt
*\*\          - RCC_INT_HSERDIF    HSE ready interrupt
*\*\          - RCC_INT_LSECSSIF   LSE CSS interrupt
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
**/
void RCC_ConfigInt(uint32_t RccInt, FunctionalState Cmd)
{
    if (Cmd != DISABLE)
    {
        /* Perform Byte access to RCC_CLKINT bits to enable the selected interrupts */
        RCC->CLKINT |= (((uint32_t)RccInt) << (uint32_t)8U);
    }
    else
    {
        /* Perform Byte access to RCC_CLKINT bits to disable the selected interrupts */
        RCC->CLKINT &= (~(((uint32_t)RccInt) << (uint32_t)8U));
    }
}

/**
*\*\name    RCC_ConfigLpuartClk.
*\*\fun     Configures the LPUART clock (LPUARTCLK).
*\*\param   RCC_LPUARTx :
*\*\          - RCC_LPUART1CLK    LPUART1
*\*\          - RCC_LPUART2CLK    LPUART2
*\*\param   RCC_LPUARTCLKSource :
*\*\          - RCC_LPUARTCLK_SRC_APB1      APB1 clock selected as LPUART clock
*\*\          - RCC_LPUARTCLK_SRC_LSI_LSE   LSI_LSE selected as LPUART clock
*\*\return  none
**/
void RCC_ConfigLpuartClk(uint32_t RCC_LPUARTx, uint32_t RCC_LPUARTCLKSource)
{
    uint32_t tmpregister;

    tmpregister = RCC->LSCTRL;
    
    /* LPUART1 */
    if(RCC_LPUARTx == RCC_LPUART1CLK)
    {
        /* Clear the LPUART1 clock source */
        tmpregister &= ~RCC_LSCTRL_LPUART1SEL;

        /* Select the LPUART1 clock source */
        tmpregister |= RCC_LPUARTCLKSource;
    }
    /* LPUART2 */
    else if(RCC_LPUARTx == RCC_LPUART2CLK)
    {
        /* Clear the LPUART2 clock source */
        tmpregister &= ~RCC_LSCTRL_LPUART2SEL;

        /* Select the LPUART2 clock source */
        tmpregister |= (RCC_LPUARTCLKSource << (uint32_t)3U);
    }
    else
    {
    	/* No Process */
    }

    /* Store the new value */
    RCC->LSCTRL = tmpregister;
}

/**
*\*\name    RCC_GetLpuartClkSrc.
*\*\fun     Returns the clock source used as LPUART clock.
*\*\param   RCC_LPUARTx :
*\*\          - RCC_LPUART1CLK    LPUART1
*\*\          - RCC_LPUART2CLK    LPUART2
*\*\return  The clock source used as LPUART clock:
 *\*\         - 0x0: APB1 used as LPUART clock
 *\*\         - 0x1: LSI_LSE used as LPUART clock
**/
uint32_t RCC_GetLpuartClkSrc(uint32_t RCC_LPUARTx)
{
	uint32_t tmp_value = 0U;

    if(RCC_LPUARTx == RCC_LPUART1CLK)
    {
    	tmp_value = ((uint32_t)(RCC->LSCTRL & RCC_LSCTRL_LPUART1SEL)>> (uint32_t)26U);
    }
    else /* RCC_LPUART2CLK */
    {
    	tmp_value = ((uint32_t)(RCC->LSCTRL & RCC_LSCTRL_LPUART2SEL)>> (uint32_t)29U);
    }

    return tmp_value;
}

/**
*\*\name    RCC_EnableLpuartClk.
*\*\fun     Enables or disables the LPUART clock.
*\*\param   RCC_LPUARTx :
*\*\          - RCC_LPUART1CLK    LPUART1
*\*\          - RCC_LPUART2CLK    LPUART2
*\*\param   Cmd :
*\*\          - ENABLE
*\*\          - DISABLE
*\*\return  none
*\*\note    This function must be used only after the LPUART clock was selected using the RCC_ConfigLpuartClk function.
**/
void RCC_EnableLpuartClk(uint32_t RCC_LPUARTx, FunctionalState Cmd)
{
    if(Cmd == ENABLE)
    {
        if(RCC_LPUARTx == RCC_LPUART1CLK)
        {
            /* Set LPUART1EN bit */
            RCC->LSCTRL |= RCC_LSCTRL_LPUART1EN;
        }
        else /* RCC_LPUART2CLK */
        {
            /* Set LPUART2EN bit */
            RCC->LSCTRL |= RCC_LSCTRL_LPUART2EN;
        }
    }
    else
    {
        if(RCC_LPUARTx == RCC_LPUART1CLK)
        {
            /* Reset LPUART1EN bit */
            RCC->LSCTRL &= ~RCC_LSCTRL_LPUART1EN;
        }
        else /* RCC_LPUART2CLK */
        {
            /* Reset LPUART2EN bit */
            RCC->LSCTRL &= ~RCC_LSCTRL_LPUART2EN;
        }
    }
}

/**
*\*\name    RCC_ConfigLse.
*\*\fun     Configures the External Low Speed oscillator (LSE).
*\*\param   RCC_LSE :
*\*\          - RCC_LSE_DISABLE    LSE oscillator OFF
*\*\          - RCC_LSE_ENABLE     LSE oscillator ON
*\*\          - RCC_LSE_BYPASS     LSE oscillator bypassed with external clock
*\*\return  none
**/
void RCC_ConfigLse(uint32_t RCC_LSE)
{
    uint32_t temp_value;
    
    temp_value = RCC->LSCTRL;
    /* Reset LSEEN bit */
    temp_value &= (~RCC_LSE_ENABLE);
    /* Reset LSEBP bit */
    temp_value &= (~RCC_LSE_BYPASS);
    /* Configure LSE (RC_LSE_DISABLE is already covered by the code section above) */
    if(RCC_LSE == RCC_LSE_ENABLE)
    {
        /* Set LSEEN bit */
        temp_value |= RCC_LSE_ENABLE;
    }   
    else if (RCC_LSE == RCC_LSE_BYPASS)
    {
        /* Set LSEBP and LSEEN bits */
        temp_value |= RCC_LSE_BYPASS;
    }
    else
    {
        /* No process */
    }

    RCC->LSCTRL = temp_value;

}

/**
*\*\name    RCC_GetClocksFreqValue.
*\*\fun     Returns the frequencies of different on chip clocks.
*\*\param   RCC_Clocks : pointer to a RCC_ClocksType structure which will hold the clocks frequencies.
*\*\return  none
*\*\note    The result of this function could be not correct when using fractional value for HSE crystal.
**/
void RCC_GetClocksFreqValue(RCC_ClocksType* RCC_Clocks)
{
    uint32_t tmp = 0U, presc = 0U;
    static const uint8_t s_ApbPresTable[8] = {0U, 0U, 0U, 0U, 1U, 2U, 3U, 4U};
    static const uint8_t s_AhbPresTable[4] = {0U, 0U, 1U, 2U};

    /* Get SYSCLK source */
    tmp = RCC->CFG & RCC_CFG_SCLKSTS;

    switch (tmp)
    {
        case 0x00000000U: /* HSI used as system clock */
            RCC_Clocks->SysclkFreq = HSI_VALUE;
            break;
        case 0x00000004U: /* HSE used as system clock */
            RCC_Clocks->SysclkFreq = HSE_VALUE;
            break;
        default:
            RCC_Clocks->SysclkFreq = HSI_VALUE;
            break;
    }

    /* Compute HCLK, PCLK1, PCLK2 and ADCCLK clocks frequencies ----------------*/
    /* Get HCLK prescaler */
    tmp   = RCC->CFG & RCC_CFG_AHBPRES;
    tmp   = tmp >> (uint32_t)4U;

    presc = s_AhbPresTable[tmp];
    /* HCLK clock frequency */
    RCC_Clocks->HclkFreq = RCC_Clocks->SysclkFreq >> presc;
    /* Get PCLK1 prescaler */
    tmp   = RCC->CFG & RCC_CFG_APB1PRES;
    tmp   = tmp >> (uint32_t)8U;
    presc = s_ApbPresTable[tmp];
    /* PCLK1 clock frequency */
    RCC_Clocks->Pclk1Freq = RCC_Clocks->HclkFreq >> presc;

    /* Get PCLK2 prescaler */
    tmp   = RCC->CFG & RCC_CFG_APB2PRES;
    tmp   = tmp >> (uint32_t)11U;
    presc = s_ApbPresTable[tmp];
    /* PCLK2 clock frequency */
    RCC_Clocks->Pclk2Freq = RCC_Clocks->HclkFreq >> presc;

    /* HSI/2 is the ADC clock source */
    if((RCC->CFG2 & RCC_CFG2_ADCSEL) == RCC_CFG2_ADCSEL)
    {
        RCC_Clocks->AdcclkFreq = HSI_VALUE / 2U;
    }
    /* HSE is the ADC clock source */
    else
    {
        RCC_Clocks->AdcclkFreq = HSE_VALUE;
    }
}

/**
*\*\name    RCC_GetFlagStatus.
*\*\fun     Checks whether the specified RCC flag is set or not.
*\*\param   RCC_FLAG :
*\*\          - RCC_CTRL_FLAG_HSIRDF       HSI oscillator clock ready
*\*\          - RCC_CTRL_FLAG_HSERDF       HSE oscillator clock ready
*\*\          - RCC_LSCTRL_FLAG_LSIRD      LSI oscillator clock ready
*\*\          - RCC_LSCTRL_FLAG_LSERD      LSE oscillator clock ready
*\*\          - RCC_LSCTRL_FLAG_LSECSSF    LSE CSS flag
*\*\          - RCC_CTRLSTS_FLAG_PINRSTF   Pin reset
*\*\          - RCC_CTRLSTS_FLAG_PORRSTF   POR reset
*\*\          - RCC_CTRLSTS_FLAG_SFTRSTF   Software reset
*\*\          - RCC_CTRLSTS_FLAG_IWDGRSTF  Independent Watchdog reset
*\*\          - RCC_CTRLSTS_FLAG_WWDGRSTF  Window Watchdog reset
*\*\return  FlagStatus (SET or RESET)
**/
FlagStatus RCC_GetFlagStatus(uint8_t RCC_FLAG)
{
    uint32_t tmp         = 0U;
    uint32_t statusreg   = 0U;
    FlagStatus bitstatus = RESET;
    
    /* Get the RCC register index */
    tmp = (uint32_t)(RCC_FLAG >> (uint32_t)5U);
    if (tmp == 1U) /* The flag to check is in CTRL register */
    {
        statusreg = RCC->CTRL;
    }
    else if (tmp == 2U) /* The flag to check is in LSCTRL register */
    {
        statusreg = RCC->LSCTRL;
    }
    else /* The flag to check is in CTRLSTS register */
    {
        statusreg = RCC->CTRLSTS;
    }

    /* Get the flag position */
    tmp = (uint32_t)(RCC_FLAG & RCC_FLAG_MASK);
    if ((statusreg & ((uint32_t)1U << tmp)) != (uint32_t)RESET)
    {
        bitstatus = SET;
    }
    else
    {
        bitstatus = RESET;
    }

    /* Return the flag status */
    return bitstatus;
}

/**
*\*\name    RCC_ClrFlag.
*\*\fun     Clears the RCC reset flags.
*\*\param   none
*\*\return  none
*\*\note    The reset flags are: 
*\*\        RCC_CTRLSTS_FLAG_PINRSTF 
*\*\        RCC_CTRLSTS_FLAG_PORRSTF 
*\*\        RCC_CTRLSTS_FLAG_SFTRSTF 
*\*\        RCC_CTRLSTS_FLAG_IWDGRSTF 
*\*\        RCC_CTRLSTS_FLAG_WWDGRSTF
**/
void RCC_ClrFlag(void)
{
    /* Set RMVF bit to clear the reset flags */
    RCC->CTRLSTS |= RCC_CTRLSTS_RMRSTF;
    /* RMVF bit should be reset */
    RCC->CTRLSTS &= ~RCC_CTRLSTS_RMRSTF;
}
