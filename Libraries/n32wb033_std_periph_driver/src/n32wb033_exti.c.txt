/**
*\*\name    EXTI_DeInit.
*\*\fun     Reset the EXTI registers to their default reset values.
*\*\return  none
**/
void EXTI_DeInit(void)
{
    EXTI->IMASK &= ~EXTI_LINEALL;
    EXTI->EMASK &= ~EXTI_LINEALL;
    EXTI->RT_CFG &= ~EXTI_LINEALL;
    EXTI->FT_CFG &= ~EXTI_LINEALL;
    EXTI->SWIE &= ~EXTI_LINEALL;
    EXTI->PEND = EXTI_LINEALL;
}

/**
*\*\name    EXTI_InitPeripheral.
*\*\fun     Initializes the EXTI peripheral according to the specified parameters in EXTI_InitStruct.
*\*\param   EXTI_InitParam(The input parameters must be the following values) :
*\*\          - EXTI_Line
*\*\            - EXTI_LINE0
*\*\            - EXTI_LINE1
*\*\            - EXTI_LINE2
*\*\            - EXTI_LINE3
*\*\            - EXTI_LINE4
*\*\            - EXTI_LINE5
*\*\            - EXTI_LINE6
*\*\            - EXTI_LINE7
*\*\            - EXTI_LINE8
*\*\            - EXTI_LINE9
*\*\            - EXTI_LINE10
*\*\            - EXTI_LINE11
*\*\            - EXTI_LINE12
*\*\            - EXTI_LINE13
*\*\            - EXTI_LINE14
*\*\            - EXTI_LINE15
*\*\            - EXTI_LINE16
*\*\            - EXTI_LINE17
*\*\            - EXTI_LINE18
*\*\            - EXTI_LINE19
*\*\            - EXTI_LINE20
*\*\            - EXTI_LINE21
*\*\          - EXTI_Mode
*\*\            - EXTI_Mode_Interrupt
*\*\            - EXTI_Mode_Event
*\*\          - EXTI_Trigger
*\*\            - EXTI_Trigger_Rising
*\*\            - EXTI_Trigger_Falling
*\*\            - EXTI_Trigger_Rising_Falling
*\*\          - EXTI_LineCmd:
*\*\            - ENABLE
*\*\            - DISABLE
*\*\return  none
**/
void EXTI_InitPeripheral(const EXTI_InitType* EXTI_InitParam)
{
    uint32_t tmp;

    tmp = (uint32_t)EXTI_BASE;

    if (EXTI_InitParam->EXTI_LineCmd != DISABLE)
    {
        /* Clear EXTI line configuration */
        EXTI->IMASK &= ~EXTI_InitParam->EXTI_Line;
        EXTI->EMASK &= ~EXTI_InitParam->EXTI_Line;

        tmp += (uint32_t)EXTI_InitParam->EXTI_Mode;

        *((__IO uint32_t*)(uintptr_t)tmp) |= EXTI_InitParam->EXTI_Line;

        /* Clear Rising Falling edge configuration */
        EXTI->RT_CFG &= ~EXTI_InitParam->EXTI_Line;
        EXTI->FT_CFG &= ~EXTI_InitParam->EXTI_Line;

        /* Select the trigger for the selected external interrupts */
        if (EXTI_InitParam->EXTI_Trigger == EXTI_Trigger_Rising_Falling)
        {
            /* Rising Falling edge */
            EXTI->RT_CFG |= EXTI_InitParam->EXTI_Line;
            EXTI->FT_CFG |= EXTI_InitParam->EXTI_Line;
        }
        else
        {
            tmp = (uint32_t)EXTI_BASE;
            tmp += (uint32_t)EXTI_InitParam->EXTI_Trigger;

            *((__IO uint32_t*)(uintptr_t)tmp) |= EXTI_InitParam->EXTI_Line;
        }
    }
    else
    {
        tmp += (uint32_t)EXTI_InitParam->EXTI_Mode;

        /* Disable the selected external lines */
        *((__IO uint32_t*)(uintptr_t)tmp) &= ~EXTI_InitParam->EXTI_Line;
    }
}

/**
*\*\name    EXTI_InitStruct.
*\*\fun     Fills each EXTI_InitStruct member with its default value.
*\*\param   EXTI_InitParam :
*\*\            - - Pointer to the EXTI_InitType structure which will be initialized.
*\*\return  none
**/
void EXTI_InitStruct(EXTI_InitType* EXTI_InitParam)
{
    EXTI_InitParam->EXTI_Line    = EXTI_LINENONE;
    EXTI_InitParam->EXTI_Mode    = EXTI_Mode_Interrupt;
    EXTI_InitParam->EXTI_Trigger = EXTI_Trigger_Falling;
    EXTI_InitParam->EXTI_LineCmd = DISABLE;
}

/**
*\*\name    EXTI_GetStatusFlag.
*\*\fun     Get EXTI line flag status.
*\*\param   EXTI_Line (The input parameters must be the following values) :
*\*\            - EXTI_LINE0
*\*\            - EXTI_LINE1
*\*\            - EXTI_LINE2
*\*\            - EXTI_LINE3
*\*\            - EXTI_LINE4
*\*\            - EXTI_LINE5
*\*\            - EXTI_LINE6
*\*\            - EXTI_LINE7
*\*\            - EXTI_LINE8
*\*\            - EXTI_LINE9
*\*\            - EXTI_LINE10
*\*\            - EXTI_LINE11
*\*\            - EXTI_LINE12
*\*\            - EXTI_LINE13
*\*\            - EXTI_LINE14
*\*\            - EXTI_LINE15
*\*\            - EXTI_LINE16
*\*\            - EXTI_LINE17
*\*\            - EXTI_LINE18
*\*\            - EXTI_LINE19
*\*\            - EXTI_LINE20
*\*\            - EXTI_LINE21
*\*\return  SET or RESET
**/
FlagStatus EXTI_GetStatusFlag(uint32_t EXTI_Line)
{
    FlagStatus bitstatus;

    if ((EXTI->PEND & EXTI_Line) != (uint32_t)RESET)
    {
        bitstatus = SET;
    }
    else
    {
        bitstatus = RESET;
    }
    return bitstatus;
}

/**
*\*\name    EXTI_GetITStatus.
*\*\fun     GET EXTI line interrupt status.
*\*\param   EXTI_Line :
*\*\            - EXTI_LINE0
*\*\            - EXTI_LINE1
*\*\            - EXTI_LINE2
*\*\            - EXTI_LINE3
*\*\            - EXTI_LINE4
*\*\            - EXTI_LINE5
*\*\            - EXTI_LINE6
*\*\            - EXTI_LINE7
*\*\            - EXTI_LINE8
*\*\            - EXTI_LINE9
*\*\            - EXTI_LINE10
*\*\            - EXTI_LINE11
*\*\            - EXTI_LINE12
*\*\            - EXTI_LINE13
*\*\            - EXTI_LINE14
*\*\            - EXTI_LINE15
*\*\            - EXTI_LINE16
*\*\            - EXTI_LINE17
*\*\            - EXTI_LINE18
*\*\            - EXTI_LINE19
*\*\            - EXTI_LINE20
*\*\            - EXTI_LINE21
*\*\return  SET or RESET
**/
INTStatus EXTI_GetITStatus(uint32_t EXTI_Line)
{
    INTStatus bitstatus;
    uint32_t enablestatus;

    enablestatus = EXTI->IMASK & EXTI_Line;
    if (((EXTI->PEND & EXTI_Line) != (uint32_t)RESET) && (enablestatus != (uint32_t)RESET))
    {
        bitstatus = SET;
    }
    else
    {
        bitstatus = RESET;
    }
    return bitstatus;
}
