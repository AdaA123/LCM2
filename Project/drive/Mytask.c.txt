
/***********************************************************************
 * @file    Mytask.c
 * @brief   轻量级定时任务调度器
 * @details 由 GTIM1 硬件更新中断提供全局 1ms 时基。应用通过 TaskAdd
 *          注册周期任务（可指定周期），主循环中调用 TaskRun 依次执行。
 *          TaskRun 会自动跳过空任务，防止无任务时取模异常。
 *          TaskGetTick 仅提供只读毫秒计数，不阻塞。
 ***********************************************************************/

#include "Mytask.h"
#include "led.h"
#include "n32wb033.h"
#include "n32wb033_rcc.h"
#include "n32wb033_tim.h"
#include "misc.h"
#include "global_func.h"

TaskType Task;

unsigned short TaskFind(void (*fun)(void))
{
    for(unsigned short i = 0; i < Task.Number; i++)
    {
        if(Task.fun[i] == fun)
            return i;
    }

    return 0XFFFF;
}

unsigned char TaskAdd(unsigned int time, void (*fun)(void)) //增加任务
{
    if(Task.Number >= TASK_NUM) return 0;

    unsigned short funPoint = TaskFind(fun);

    if(funPoint == 0XFFFF)
    {
        Task.Count[Task.Number] = time;
        Task.fun[Task.Number++] = fun;
    }
    else
        Task.Count[funPoint] = time;

    return 1;
}

unsigned char TaskDelete(void (*fun)(void)) //删除任务
{
    unsigned short point = TaskFind(fun);

    if(point != 0XFFFF)
    {
        Task.fun[point] = Task.fun[Task.Number - 1];
        Task.Count[point] = Task.Count[Task.Number - 1];
        Task.Number--;
        return 1;
    }

    return 0;
}

unsigned int TaskTime = 0;
unsigned short ActionTypeTasT = 0;

void TaskRun(void) //系统定时任务
{
    static unsigned short ActionTypeNum = 0;

    if(Task.Time && Task.Number)
    {
        if(TaskTime % Task.Count[ActionTypeNum] == 0)
            Task.fun[ActionTypeNum]();

        if(++ActionTypeNum >= Task.Number)
        {
            ActionTypeNum = 0;
            Task.Time--;
            TaskTime++;
        }
    }
}

void TaskTimeAdd(void)
{
    Task.Time++;
}

void TaskInit(void (*fun)(void))
{
    Task.ActionTypeTask = fun;
    Task.Time = 0;
}

/* GTIM1 1ms更新中断：清除挂起标志后，让 Task.Time 加1 */
static volatile uint32_t s_tick_ms = 0u;   /* 真实毫秒计数（只在中断里更新） */

void TaskTick_IRQHandler(void)
{
    if(TIM_GetIntStatus(GTIM1, TIM_INT_UPDATE) != RESET)
    {
        TIM_ClrIntPendingBit(GTIM1, TIM_INT_UPDATE);
        s_tick_ms++;
        TaskTimeAdd();
    }
}

/* 只读获取真实毫秒计数，不阻塞 */
uint32_t TaskGetTick(void)
{
    return s_tick_ms;
}

/* 启动系统1ms节拍：必须严格按照下面的顺序初始化，提前触发中断会导致死机 */
void TaskStart(void)
{
    TIM_TimeBaseInitType tb;
    NVIC_InitType nvic;
    /* step1: 打开GTIM1外设时钟 */
    RCC_EnableAPB1PeriphClk(RCC_APB1_PERIPH_GTIM1, ENABLE);
    /* step2: 配置1ms周期时基（计数频率1MHz，自动重装载1000） */
    TIM_InitTimBaseStruct(&tb);
    tb.Period      = (1000000u / 1000u) - 1u;
    tb.Prescaler   = (uint16_t)((SystemCoreClock / 1000000u) - 1u);
    tb.ClkDiv      = TIM_CLK_DIV1;
    tb.CounterMode = TIM_CNT_MODE_UP;
    TIM_InitTimeBase(GTIM1, &tb);
    TIM_ConfigPrescaler(GTIM1, tb.Prescaler, TIM_PSC_RELOAD_MODE_IMMEDIATE);
    /* step3: 注册中断处理函数到ROM中断向量表 */
    ModuleIrqRegister(GTIM1_IRQn, TaskTick_IRQHandler);
    /* step4: 清除外设与NVIC中挂起的中断标志 */
    TIM_ClrIntPendingBit(GTIM1, TIM_INT_UPDATE);
    NVIC_ClearPendingIRQ(GTIM1_IRQn);
    /* step5: 配置NVIC中断 */
    nvic.NVIC_IRQChannel         = GTIM1_IRQn;
    nvic.NVIC_IRQChannelPriority = NVIC_PRIORITY_3;
    nvic.NVIC_IRQChannelCmd      = ENABLE;
    NVIC_Init(&nvic);
    /* step6: 使能更新中断并启动定时器 */
    TIM_ConfigInt(GTIM1, TIM_INT_UPDATE, ENABLE);
    TIM_Enable(GTIM1, ENABLE);
}
