#define PROTO_TARGET_NAME_SIZE 24u

/* 0x81 屏数据应答 reply[3]：屏与 BMS 的连接状态。 */
typedef enum
{
    PROTO_BMS_DISCONNECTED = 0,      /* 未连接电池 / 已解绑 */
    PROTO_BMS_CONNECTED = 1,         /* BMS 数据通道已就绪 */
    PROTO_BMS_CONNECTING = 2,        /* 等待连接 / 服务发现中 */
    PROTO_BMS_CONNECTION_FAILED = 3  /* 本轮连接超时失败，已停止，等待新请求 */
} proto_bms_connection_state_t;

typedef struct

    Int16 (*bms_peer_get)(proto_peer_info_t *peer); /* 连接中优先返回目标，否则读取当前对端。 */
    Byte (*bms_connection_state)(void); 
    UInt16 (*local_project_no_get)(void);
    Int16 (*local_identity_set)(const Byte *name, UInt16 length, const Byte mac_msb[6]);
