i < len - 1; i++) /* 外循环为排序趟数，len个数进行len-1趟 */    {        for(j = 0; j < len - 1 - i; j++) /* 内循环为每趟比较的次数，第i趟比较len-i次 */        {            if(arr[j] > arr[j + 1])  /* 相邻元素比较，若逆序则交换（升序为左大于右，降序反之）*/            {                temp = arr[j];                arr[j] = arr[j + 1];                arr[j + 1] = temp;            }        }    }}void ByteFromSmallBig(Byte *arr, UInt32 len)  //Byte数据从大到小排列{    int i, j, temp;    for(i = 0;
