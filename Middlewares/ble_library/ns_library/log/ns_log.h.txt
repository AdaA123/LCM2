/***********************************************************************
 * @file    ns_log.h
 * @brief   Compile-time log level macros
 ***********************************************************************/
#ifndef NS_LOG_H
#define NS_LOG_H

#include <stdio.h>
#include "app_user_config.h"

#ifdef __cplusplus
extern "C" {
#endif

#ifndef NS_LOG_ERROR_ENABLE
#define NS_LOG_ERROR_ENABLE 0
#endif
#ifndef NS_LOG_WARNING_ENABLE
#define NS_LOG_WARNING_ENABLE 0
#endif
#ifndef NS_LOG_INFO_ENABLE
#define NS_LOG_INFO_ENABLE 0
#endif
#ifndef NS_LOG_DEBUG_ENABLE
#define NS_LOG_DEBUG_ENABLE 0
#endif
#ifndef PRINTF_COLOR_ENABLE
#define PRINTF_COLOR_ENABLE 0
#endif

#define LOG_COLOR_RED     "\033[0;31m"
#define LOG_COLOR_YELLOW  "\033[0;33m"
#define LOG_COLOR_CYAN    "\033[0;36m"
#define LOG_COLOR_GREEN   "\033[0;32m"

#define NS_LOG_OUTPUT(color, ...)          \
do {                                       \
    if (PRINTF_COLOR_ENABLE) {             \
        printf("%s", (color));             \
    }                                      \
    printf(__VA_ARGS__);                   \
} while (0)

#if NS_LOG_ERROR_ENABLE
#define NS_LOG_ERROR(...) NS_LOG_OUTPUT(LOG_COLOR_RED, __VA_ARGS__)
#else
#define NS_LOG_ERROR(...) do { } while (0)
#endif

#if NS_LOG_WARNING_ENABLE
#define NS_LOG_WARNING(...) NS_LOG_OUTPUT(LOG_COLOR_YELLOW, __VA_ARGS__)
#else
#define NS_LOG_WARNING(...) do { } while (0)
#endif

#if NS_LOG_INFO_ENABLE
#define NS_LOG_INFO(...) NS_LOG_OUTPUT(LOG_COLOR_CYAN, __VA_ARGS__)
#else
#define NS_LOG_INFO(...) do { } while (0)
#endif

#if NS_LOG_DEBUG_ENABLE
#define NS_LOG_DEBUG(...) NS_LOG_OUTPUT(LOG_COLOR_GREEN, __VA_ARGS__)
#else
#define NS_LOG_DEBUG(...) do { } while (0)
#endif

#ifdef __cplusplus
}
#endif

#endif /* NS_LOG_H */
