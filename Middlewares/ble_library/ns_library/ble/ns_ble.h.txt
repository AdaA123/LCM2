/**
 * @brief The ble callback message environment structure for Application.
 */
struct ble_msg_t
{
    enum app_ble_msg msg_id;

    //income message
    union
    {
        struct gapm_cmp_evt const              *p_gapm_cmp;
        struct gapc_connection_req_ind const   *p_connection_ind;
        struct gapc_disconnect_ind const       *p_disconnect_ind;
        struct gapc_param_update_req_ind const *p_param_req;
        struct gapc_cmp_evt const              *p_gapc_cmp;
        struct gapc_con_rssi_ind const         *p_gapc_rssi;
        struct gattc_mtu_changed_ind const     *p_gattc_mtu;
        struct gapc_param_updated_ind const    *p_param_updated;
        struct gattc_cmp_evt const             *p_gattc_cmp;
    } msg;
    // output command
    union
    {
        struct gapm_set_dev_config_cmd         *p_dev_config;
        struct gapc_connection_cfm             *p_connection_cfm;
        struct gapc_param_update_cfm           *p_param_cfm;
    } cmd;
    /// Connection index associated with a connection-scoped event
    uint8_t conn_index;
};

/**

{
    uint16_t conhdl;    /**< Connection handle */

 * @brief Application environment structure.

    struct bd_addr peer_addr;           /**< Latest connected peer address */

    struct gap_sec_key peer_irk;    /**< peer device IRK */
    uint8_t rand_cnt;                           /**< Counter used to generate IRK */
    uint16_t max_mtu;                           /**< Maximum device MTU size */
    uint16_t tx_pkt_size;                   /**< DLE, the maximum number of payload octets in TX */

    //master device
    uint8_t scan_actv_idx;              /**< Scan activity index */
    uint8_t init_actv_idx;              /**< Init activity index */
    uint8_t target_found;                   /**< flag for founding the target nearby */
    uint8_t target_addr_type;           /**< target device address type */
    struct bd_addr target_addr;     /**< target device address going to connect */

    void (*ble_msg_handler)(struct ble_msg_t const *);                  /**< ble msg hanlder */
    void (*user_msg_handler)(ke_msg_id_t const, void const *);  /**< user msg handler */

#if (BLE_APP_NS_IUS)
    uint8_t manual_conn_param_update;       /**< flag for connection paramter update in OTA */
    uint8_t manual_mtu_update;                  /**< flag for mtu update in OTA */
#endif
    struct app_con_env_tag conn_env[APP_CON_IDX_MAX];       /**< Connection environment structure for Application */

    uint32_t rssi_intv;         /**< the interval for getting rssi */
    uint8_t  rssi_conidx;       /**< the conidx for getting rssi */
    ble_lsc_cfg_t lsc_cfg;  /**< The Low-speed clock type */
    bool  wl_reconnect;         /**< the flag for reconnect with white list  */
};

/**
 * @brief Scan environment structurein in Application.
 */
struct ns_scan_params_t
{
    uint8_t type;                                           /**< Scanning Type (@see enum gapm_scan_type) */
    uint8_t scan_enable         : 1;  /**< if enable scan */
    uint8_t connect_enable      : 1;  /**< if connect the device if match */
    uint8_t phy_coded_enable    : 1;  /**< if use coded phy mode */
    uint8_t prop_active_enable  : 1;    /**< Properties for the scan active mode, require response data */

    void (*ble_scan_state_handler)(enum scan_state_t);                                          /**< scan state callback function */
    void (*ble_scan_data_handler)(struct gapm_ext_adv_report_ind const*);       /**< scan data callback function */

/**
 * @brief Enumeration of TX/RX PHY values.
 */
enum ble_phy_val
{
    BLE_PHY_ANY          = GAP_PHY_ANY,             /**< No preferred PHY */
    BLE_PHY_1MBPS        = GAP_PHY_LE_1MBPS,    /**< LE 1M PHY (TX or RX) */
    BLE_PHY_2MBPS        = GAP_PHY_LE_2MBPS,    /**< LE 2M PHY (TX or RX) */
    BLE_PHY_CODED        = GAP_PHY_LE_CODED,    /**< LE Coded PHY (RX Only) */
    BLE_PHY_500KBPS      = GAP_PHY_LE_CODED | (GAPC_PHY_OPT_LE_CODED_500K_RATE << 4), /**< LE Coded PHY with S=8 data coding (TX Only) */
    BLE_PHY_125KBPS      = GAP_PHY_LE_CODED | (GAPC_PHY_OPT_LE_CODED_125K_RATE << 4), /**< LE Coded PHY with S=2 data coding (TX Only) */
};
