/* Public define ------------------------------------------------------------*/	
/// Branch configuration for patch control
typedef struct
{
    /// Patch for ke_timer_set function
    bool patch_ke_timer_set  : 1;
    /// Patch for 3 connection support
    bool patch_3_conn : 1;
    /// Patch for advertisement interval check
    bool patch_adv_interval : 1;
    /// Patch for duration calculation 1
    bool patch_duration1 : 1;
    /// Patch for duration calculation 2
    bool patch_duration2 : 1;
    /// Default pin timer flag (for llc_le_ping_set)
    bool default_pin_timer : 1;
    /// Default PHY update procedure flag
    bool default_phy_upd_proc : 1;
    /// Default data length update procedure flag
    bool default_dl_upd_proc : 1;
    /// Clear authenticated payload timers in llc_cleanup
    bool llc_cleanup_timer_clear : 1;
    /// Force gen private key for ECC operations
    bool force_gen_private_key : 1;
    /// Restrict remote supported features based on version number
    bool restrict_rem_supp_feat : 1;

    /// App init branch patch
    bool app_init_branch : 1;
    /// GAPM setup init branch patch
    bool gapm_setup_init : 1;
    /// GAPM init attribute database branch patch
    bool gapm_init_attr : 1;
    /// GATTM init attribute database branch patch
    bool gattm_init_attr : 1;
    /// GAPM set local address branch patch (true: use GAPM_PRIV_CFG_PRIV_EN_BIT check, false: use gapc_is_local_addr_rpa check)
    bool gapm_set_local_addr : 1;
    /// HCI transport on/off branch patch
    bool hci_transport_onff : 1;
    /// AHI init branch patch (true: call ahi_init, false: skip)
    bool ahi_init_branch : 1;
    /// NS corrself config branch patch
    bool ns_corrself_cfg : 1;
    /// Enable llhwc_phy_prerx in sleep isr (true: enable, false: disable)
    bool enable_prerx_in_slp : 1;
    /// LLD connection branch patch 0
    bool lld_con_branch0 : 1;
    /// LLD connection branch patch 1
    bool lld_con_branch1 : 1;
    /// LLD connection branch patch 2
    bool lld_con_branch2 : 1;
    /// LLD connection branch patch 3
    bool lld_con_branch3 : 1;
    /// LLD connection branch patch 4
    bool lld_con_branch4 : 1;
    /// LLD connection branch patch 5
    bool lld_con_branch5 : 1;
    /// LLD connection branch patch 6
    bool lld_con_branch6 : 1;
    /// LLD connection branch patch 7
    bool lld_con_branch7 : 1;
    /// LLD connection branch patch 8
    bool lld_con_branch8 : 1;
    /// LLD connection branch patch 9
    bool lld_con_branch9 : 1;
    /// LLD connection branch patch 10
    bool lld_con_branch10 : 1;
    /// Reserved bits for future patches
    uint32_t reserved : 1;
} branch_cfg_t;

typedef struct
{
    void (*init)(uint8_t init_type);
    void (*create)(uint8_t conidx);
    void (*cleanup)(uint8_t conidx, uint8_t reason);
    uint8_t (*add_profile)(void *params, uint16_t *prf_task);
    uint16_t (*get_task_from_id)(uint16_t id);
    uint16_t (*get_id_from_task)(uint16_t task);
} prf_func_tbl_t;

/** Flash operation return codes **/
typedef enum {
    FlashOperationSuccess,    /* Operation completed successfully            */
    FlashWriteRegFailed,      /* Failed to write status/configuration register */
    FlashTimeOut,             /* Operation timed out                         */
    FlashIsBusy,              /* Flash is busy (WIP=1)                       */
    FlashQuadNotEnable,       /* Quad mode not enabled                       */
    FlashAddressInvalid,      /* Address out of range                        */
} ReturnMsg;
// NOR Flash information structure
// Contains configuration details for different NOR Flash devices
typedef struct {
    uint32_t flash_id;      // Unique Flash device ID
    const char *vendor;     // Vendor name
    const char *model;      // Device model name
    uint8_t qspi_mode;      // QSPI mode (MODE_4LINE_4CYCLE/MODE_4LINE_2CYCLE/MODE_2LINE)
    uint16_t flash_size;    // Flash size in KB
    uint8_t otp_count;      // Number of OTP blocks
    uint16_t otp_size;      // Size of each OTP block in bytes
    uint16_t otp_1st_addr;  // Address of 1st OTP block (OTPINV if not present)
    uint16_t otp_2nd_addr;  // Address of 2nd OTP block (OTPINV if not present)
    uint16_t otp_3rd_addr;  // Address of 3rd OTP block (OTPINV if not present)
    uint16_t otp_4th_addr;  // Address of 4th OTP block (OTPINV if not present)
    uint8_t status_reg_type;// Status register type (STATUS_REG_ONE/STATUS_REG_TWO)
} nor_flash_info;
